\section{Relación de Orden en $\Z$}
% ============================================================

\begin{enumerate}[leftmargin=*]
  \item Complete con el símbolo $<$, $>$ o $=$ según corresponda:
        \[
          \text{a) } -8 \;\rule[-2pt]{1cm}{0.4pt}\; -3 \qquad \qquad
          \text{b) } 0 \;\rule[-2pt]{1cm}{0.4pt}\; -5
        \]
        \[
          \text{c) } |-9| \;\rule[-2pt]{1cm}{0.4pt}\; |3| \qquad \qquad
          \text{d) } -4 \;\rule[-2pt]{1cm}{0.4pt}\; 4
        \]

  \item Ordene de \textbf{menor a mayor} los siguientes enteros:
        \[
          -12, \quad 5, \quad -1, \quad 0, \quad -7, \quad 3
        \]

  \item En una ciudad de la sierra se registraron estas temperaturas: madrugada $-9\,^{\circ}$C, mediodía $4\,^{\circ}$C y noche $-2\,^{\circ}$C. Ordene las temperaturas de menor a mayor e indique en qué momento hizo más frío.

  \item \textquestiondown Es verdadero que $-15 > -8$? Justifique su respuesta usando el valor absoluto.
\end{enumerate}

\section{Suma y Resta en $\Z$}
% ============================================================

\begin{enumerate}[leftmargin=*]
  \item Calcule el valor exacto de:
        \[
          |-12 + 5| - |-8 - 3| + |-(-7)|
        \]

  \item Resuelva, simplificando primero los signos dobles:
        \[
          -(3 - 9) - [-(5 - 12) + 4]
        \]

  \item Obtenga el resultado de la siguiente cadena de signos:
        \[
          -(-15) + (-8) - (+6) - (-(-4))
        \]

  \item Encuentre el número entero que falta para que la igualdad sea verdadera:
        \[
          -10 + \_\_\_ - (-5) = -18
        \]
\end{enumerate}

\section{Multiplicación y División en $\Z$}
% ============================================================

\begin{enumerate}[leftmargin=*]
  \item Calcule:
        \[
          (-4) \times (-6) \div (-3) \times 2
        \]

  \item Calcule:
        \[
          (-9) \times 8 \div (-6) \div (-4)
        \]

  \item Encuentre el número que falta en cada caso:
        \begin{enumerate}[label=\alph*)]
          \item $\_\_\_ \times (-7) = 56$
          \item $(-48) \div \_\_\_ = -6$
        \end{enumerate}

  \item Primero decida el signo del resultado (contando los factores negativos) y luego calcule:
        \[
          (-5) \times (-4) \times (-3) \div (-2) \div (-5)
        \]
\end{enumerate}

\section{Operaciones Combinadas}
% ============================================================

\begin{enumerate}[leftmargin=*]
  \item Resuelva:
        \[
          -5 \times [4 - (-3) \times 2] \div (-2)
        \]

  \item Resuelva, evaluando de adentro hacia afuera:
        \[
          [-12 \div (-3) - (-4) \times 2] \div (-2) - (-1)
        \]

  \item Resuelva:
        \[
          \{[6 - (-2)] \times (-3) + (-12)\} \div (-6)
        \]

  \item Resuelva la operación multinivel:
        \[
          -2 \times \{-5 + 3 \times [-4 - (-12 \div 3 + 1)]\} - (-10)
        \]
\end{enumerate}

\section{Problemas con Números Enteros}
% ============================================================

\begin{enumerate}[leftmargin=*]
  \item En la sierra, a las 6 de la mañana la temperatura era de $-8\,^{\circ}$C. A lo largo del día subió $15\,^{\circ}$C, luego bajó $22\,^{\circ}$C, subió $7\,^{\circ}$C y finalmente bajó $3\,^{\circ}$C. \textquestiondown Cuál fue la temperatura final?

  \item Un ascensor parte del piso $-2$ (sótano). Sube $9$ pisos, baja $5$, sube $12$, baja $7$ y finalmente baja $6$. \textquestiondown En qué piso termina?

  \item Martín debía $24$ Bs. a su amigo. El lunes le pagó $9$ Bs., el martes $5$ Bs. y el miércoles $12$ Bs. Representa la deuda como un número entero y determina cuánto debe al final.

  \item Un buzo en una competencia parte de la superficie ($0$ m). Desciende $4$ metros por minuto durante $3$ minutos, luego asciende $9$ metros y después desciende $2$ metros por minuto durante $5$ minutos. \textquestiondown A qué profundidad termina?
\end{enumerate}

  Los siguientes ejercicios son \textbf{práctica adicional} de nivel medio y difícil: \emph{no forman parte del puntaje base del examen}. Úsalos para reforzar el valor absoluto, las propiedades y las operaciones combinadas de varios niveles.

\subsection*{A.\quad Suma, resta y valor absoluto}

\begin{enumerate}[leftmargin=*]
  \item $|-15| - |-9| + (-7) - (-2)$

  \item $-(-(-8)) + (-3) - (+5) + (-(-2))$

  \item Complete el número que falta:
        \[
          7 - \_\_\_ + (-12) = -4
        \]

  \item $|-9| + |4 - 11| - |-6|$

  \item $\big|\,|-15| - |8|\,\big| + |-3|$

  \item $|(-7) \cdot 4| \div |-14|$
\end{enumerate}

\subsection*{B.\quad Multiplicación, división y propiedades}

\begin{enumerate}[leftmargin=*, start=7]
  \item $(-6) \times (-8) \div (-4) \div (-3) \times (-2)$

  \item Complete el número que falta:
        \[
          (-5) \times \_\_\_ \div (-3) = -10
        \]

  \item Primero decide el signo y luego calcula:
        \[
          (-2) \times (-3) \times (-4) \div (-6) \div (-1)
        \]

  \item Verifica la igualdad y nombra la propiedad:
        \[
          [(-6) + 4] + (-9) = (-6) + [4 + (-9)]
        \]

  \item Calcula aplicando las propiedades:
        \[
          4 \cdot 37 \cdot (-25)
        \]

  \item Calcula usando la propiedad distributiva:
        \[
          (-8) \cdot 103
        \]
\end{enumerate}

\subsection*{C.\quad Operaciones combinadas y problemas}

\begin{enumerate}[leftmargin=*, start=13]
  \item $5 - \{[(-3) \cdot 4 - (-6)] \div (-2) + 8\}$

  \item $[(-2) \cdot (7 - 12) + (-15) \div 3] \cdot [(-4 + 1) \div (-3)]$

  \item $-3 \cdot \{4 - [(-8) \div (-2) + 3 \cdot (-1)]\} + (-6)$

  \item Un submarino está a $-35$ m. Asciende $18$ m, desciende $9$ m, asciende $22$ m y desciende $14$ m. \textquestiondown A qué profundidad termina?

  \item Una cámara de frío registra $-18\,^{\circ}$C. Se apaga y la temperatura sube $3\,^{\circ}$C cada hora durante $4$ horas; luego se enciende y baja $5\,^{\circ}$C cada hora durante $2$ horas. \textquestiondown Cuál es la temperatura final?

  \item Andrés tiene una deuda de $45$ Bs. Paga $18$ Bs., luego pide prestados $12$ Bs. más y finalmente paga $20$ Bs. \textquestiondown Cuál es su saldo al final?
\end{enumerate}

\subsection*{D.\quad Razonamiento}

\begin{enumerate}[leftmargin=*, start=19]
  \item Si $a$ y $b$ son enteros negativos, \textquestiondown $a + b$ es siempre negativo? Justifica con un ejemplo.

  \item \textquestiondown Se cumple la propiedad conmutativa para la resta en $\Z$? Demuéstralo con un contraejemplo.

  \item Si $|x| = 6$, \textquestiondown qué valores puede tomar $x$?

  \item Calcula $15 \cdot 99$ usando la propiedad distributiva.

  \item Ordena de \textbf{mayor a menor}: $\ 7, \ -2, \ -11, \ 0, \ 5, \ -8$.

  \item Una persona debe $18$ Bs. y otra debe $25$ Bs. Representa cada deuda como un número entero, compáralas y decide cuál es la peor situación.
\end{enumerate}

  \subsection*{Relación de Orden en $\Z$ (Ejercicios 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item a) $-8 < -3$ \quad b) $0 > -5$ \quad c) $|-9| = 9 > 3 = |3|$, luego $>$ \quad d) $-4 < 4$
    \item $-12 < -7 < -1 < 0 < 3 < 5$
    \item $-9 < -2 < 4$. Hizo más frío en la madrugada ($-9\,^{\circ}$C): entre negativos, $-9$ es el menor porque $|-9| = 9 > 2 = |-2|$.
    \item \textbf{Falso}. $-15 < -8$, porque $|-15| = 15 > 8 = |-8|$: entre dos negativos es mayor el que está más cerca del cero, es decir, el de \emph{menor} valor absoluto.
  \end{enumerate}

  \subsection*{Suma y Resta en $\Z$ (Ejercicios 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item $|-7| - |-11| + |7| = 7 - 11 + 7 = 3$
    \item $(3 - 9) = -6 \Rightarrow -(-6) = 6$; \ $(5 - 12) = -7 \Rightarrow -(-7) + 4 = 11$; \ $6 - 11 = -5$
    \item $15 - 8 - 6 - 4 = -3$
    \item $-10 + x + 5 = -18 \Rightarrow x = -13$
  \end{enumerate}

  \subsection*{Multiplicación y División en $\Z$ (Ejercicios 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item $(-4) \times (-6) = 24$; $24 \div (-3) = -8$; $-8 \times 2 = -16$
    \item $(-9) \times 8 = -72$; $-72 \div (-6) = 12$; $12 \div (-4) = -3$
    \item a) $-8$ \quad b) $8$
    \item Cinco factores negativos (cantidad impar) $\Rightarrow$ resultado negativo.
          $(-5) \times (-4) = 20$; $20 \times (-3) = -60$; $-60 \div (-2) = 30$; $30 \div (-5) = -6$
  \end{enumerate}

  \subsection*{Operaciones Combinadas (Ejercicios 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item $(-3) \times 2 = -6$; $4 - (-6) = 10$; $-5 \times 10 = -50$; $-50 \div (-2) = 25$
    \item $-12 \div (-3) = 4$; $(-4) \times 2 = -8$; $4 - (-8) = 12$; $12 \div (-2) = -6$; $-6 - (-1) = -5$
    \item $6 - (-2) = 8$; $8 \times (-3) = -24$; $-24 + (-12) = -36$; $-36 \div (-6) = 6$
    \item $(-12 \div 3 + 1) = -3$; $[-4 - (-3)] = -1$; $3 \times (-1) = -3$; $-5 + (-3) = -8$;
          $-2 \times (-8) = 16$; $16 - (-10) = 26$
  \end{enumerate}

  \subsection*{Problemas con Números Enteros (Ejercicios 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item $-8 + 15 = 7$; $7 - 22 = -15$; $-15 + 7 = -8$; $-8 - 3 = -11\,^{\circ}$C
    \item $-2 + 9 = 7$; $7 - 5 = 2$; $2 + 12 = 14$; $14 - 7 = 7$; $7 - 6 = 1$ \ (piso $1$)
    \item $-24 + 9 + 5 + 12 = 2$ \ $\Rightarrow$ la deuda queda en $0$ y le sobran $2$ Bs.
    \item Descenso inicial: $3 \times (-4) = -12$; $-12 + 9 = -3$;
          descenso final: $5 \times (-2) = -10$; $-3 - 10 = -13$ m
  \end{enumerate}

  \subsection*{Anexo: Ejercicios de Ampliación}

    \item $-18 + 4 \cdot 3 = -18 + 12 = -6$; $-6 - 2 \cdot 5 = -6 - 10 = -16\,^{\circ}$C.
    \item $-45 + 18 - 12 + 20 = -19$ \ $\Rightarrow$ debe $19$ Bs.
