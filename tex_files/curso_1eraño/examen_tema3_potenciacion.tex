          \item En $4^{3} = 64$: el $4$ se llama \rule{2cm}{0.4pt}, el $3$ se llama \rule{2cm}{0.4pt} \\ y $64$ es el valor de la \rule{2cm}{0.4pt}.
          \item El producto $5 \cdot 5 \cdot 5 \cdot 5$ se escribe como potencia: \rule{2cm}{0.4pt}.
          \item Toda base distinta de cero elevada al exponente $0$ es igual a \rule{1.5cm}{0.4pt}. \\ Por ejemplo: $9^{0} = $ \rule{1.5cm}{0.4pt}.
          \item El valor de $(-3)^{2}$ es \rule{1.5cm}{0.4pt} y el de $-3^{2}$ es \rule{1.5cm}{0.4pt}.
        \end{enumerate}

  \item \textbf{(5 pts)} \textbf{Verdadero o Falso.} Escriba \textbf{V} o \textbf{F} según corresponda y corrija las afirmaciones falsas:
        \begin{enumerate}[label=\alph*)]
          \item $3^{2} = 3 \cdot 2$.
          \item $(-4)^{2} = 16$.
          \item $-5^{2} = 25$.
          \item Si $a \neq 0$, entonces $a^{0} = 1$.
          \item $2^{3} + 2^{4} = 2^{7}$.
        \end{enumerate}

  \item \textbf{(5 pts)} \textbf{Completación: propiedades de las potencias.} Complete con el exponente o la expresión que falta (considere las bases distintas de cero):
        \begin{enumerate}[label=\alph*)]
          \item $a^{m} \cdot a^{n} = a^{\rule{1.2cm}{0.4pt}}$
          \item $\dfrac{a^{m}}{a^{n}} = a^{\rule{1.2cm}{0.4pt}}$
          \item $(a^{m})^{n} = a^{\rule{1.2cm}{0.4pt}}$
          \item $(a \cdot b)^{n} = a^{\rule{0.8cm}{0.4pt}} \cdot b^{\rule{0.8cm}{0.4pt}}$
          \item $\left(\dfrac{a}{b}\right)^{n} = \dfrac{a^{\rule{0.8cm}{0.4pt}}}{b^{\rule{0.8cm}{0.4pt}}}$
        \end{enumerate}

  \item \textbf{(5 pts)} \textbf{Definiciones breves.} Responda con sus propias palabras:

  \subsection*{Parte Teórica (Preguntas 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item a) \emph{multiplicación}. \quad b) \emph{base}, \emph{exponente}, \emph{potencia}. \quad c) $5^{4}$. \quad d) $1$; $9^{0} = 1$. \quad e) $(-3)^{2} = 9$ y $-3^{2} = -9$.

    \item a) \textbf{F}: $3^{2} = 3 \cdot 3 = 9$, no $3 \cdot 2 = 6$. \quad b) \textbf{V}: $(-4)(-4) = 16$. \quad c) \textbf{F}: $-5^{2} = -(5 \cdot 5) = -25$ (el signo queda fuera del paréntesis). \quad d) \textbf{V}. \quad e) \textbf{F}: $2^{3} + 2^{4} = 8 + 16 = 24 \neq 2^{7} = 128$ (los exponentes se suman solo al multiplicar potencias de igual base).

    \item a) $m + n$. \quad b) $m - n$. \quad c) $m \cdot n$. \quad d) $n$ y $n$. \quad e) $n$ y $n$.

    \item a) En $(-5)^{2}$ el paréntesis incluye el signo: $(-5)(-5) = 25$. En $-5^{2}$ el signo queda afuera: $-(5 \cdot 5) = -25$.
          \quad b) Consiste en escribir las bases compuestas como potencias de un mismo primo; por ejemplo, $8 = 2^{3}$, luego $8^{2} = (2^{3})^{2} = 2^{6}$.
          \quad c) Cuando la base es negativa y el exponente es impar.
  \end{enumerate}

  \subsection*{Base, Signos y Operaciones Combinadas (Ejercicios 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item $(-2)^{4} = 16$; \ $-2^{4} = -16$; \ $(-3)^{3} = -27$; \ $-(-3)^{2} = -9$
    \item $(-1)^{2025} = -1$; $(-1)^{2026} = 1$; $0^{7} = 0$; $10^{0} = 1$; $-(-5)^{1} = 5$ \ $\Rightarrow$ $-1 + 1 + 0 + 1 + 5 = 6$
    \item a) $(-7)^{12} > 0$ (exponente par); \quad b) $(-8)^{9} < 0$ (exponente impar); \quad c) $(-5)^{0} = 1 > 0$; \quad d) $(-2)^{15} < 0$ (impar), pero el signo menos de afuera lo cambia a positivo: $-(-2)^{15} > 0$.
    \item $9 + \dfrac{36 \cdot 4}{16} - 3^{3} + 8 \cdot 6^{0} = 9 + 9 - 27 + 8 = -1$
  \end{enumerate}

  \subsection*{Propiedades de las Potencias y Cambio de Base (Ejercicios 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item a) $2^{6+4+1} = 2^{11}$ \quad b) $\dfrac{9^{8+3}}{9^{6}} = 9^{11-6} = 9^{5}$ \quad c) $2^{24} \div 2^{19} = 2^{5} = 32$
    \item a) $10^{3} = 1000$ \quad b) $10^{7} \div 10^{4} = 10^{3} = 1000$ \quad c) $\left(\dfrac{12}{4}\right)^{3} + \dfrac{15^{2}}{15^{2}} = 3^{3} + 1 = 28$
    \item a) $\dfrac{2^{10} \cdot 2^{6}}{2^{12}} = \dfrac{2^{16}}{2^{12}} = 2^{4} = 16$ \quad b) $(3^{6} \cdot 3^{6})^{3} = (3^{12})^{3} = 3^{36}$ \quad c) $(2^{18} \div 2^{14})^{4} = (2^{4})^{4} = 2^{16} = 65\,536$
    \item a) $\dfrac{(2^{2} \cdot 3)^{4} \cdot (2 \cdot 3^{2})^{2}}{(2 \cdot 3)^{4} \cdot 3^{4}} = \dfrac{2^{10} \cdot 3^{8}}{2^{4} \cdot 3^{8}} = 2^{6} = 64$
          \quad b) $\dfrac{(2 \cdot 5^{2})^{3} \cdot 2^{2}}{(2 \cdot 5)^{4} \cdot 5} = \dfrac{2^{5} \cdot 5^{6}}{2^{4} \cdot 5^{5}} = 2 \cdot 5 = 10$
  \end{enumerate}

  \subsection*{Potencias con Letras — Mini Álgebra (Ejercicios 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item a) $2(-3)^{2}(2)^{3} - 3(-3)^{2}(2)^{2} = 2(9)(8) - 3(9)(4) = 144 - 108 = 36$
          \quad b) $(-2)^{3} + 3(-2)^{2} - 5(-2) + 1 = -8 + 12 + 10 + 1 = 15$
    \item a) $4p^{6} \cdot 9p^{4} = 36p^{10}$ \quad b) $8x^{3} \cdot 3(1) = 24x^{3}$ \quad c) $(9x^{2} \cdot 4x^{6})^{3} = (36x^{8})^{3} = 46\,656\,x^{24}$
    \item a) $\left(a^{4}b^{3}\right)^{3} = a^{12}b^{9}$ \quad b) $\dfrac{9m^{4}n^{6} \cdot 8m^{3}n^{3}}{6m^{5}n^{6}} = \dfrac{72m^{7}n^{9}}{6m^{5}n^{6}} = 12m^{2}n^{3}$
    \item Arriba: $m^{6} \cdot n^{6}m^{2} \cdot n^{18} = m^{8}n^{24}$; abajo: $n^{4}m^{8}$; fracción: $n^{20}$; al cubo: $(n^{20})^{3} = n^{60}$.
  \end{enumerate}

  \subsection*{Problemas de Aplicación y Razonamiento (Ejercicios 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item a) $2^{6} = 64$ bacterias \quad b) $2^{8} \div 2^{5} = 2^{3} = 8$ veces más bacterias
    \item a) Área $= x^{2} = 10^{2} = 100$ cm$^{2}$; Volumen $= y^{3} = 4^{3} = 64$ cm$^{3}$ \quad b) $(2x)^{2} = 4x^{2}$: el área se \textbf{cuadruplica}.
    \item a) $3^{n+2} = 3^{7} \Rightarrow n = 5$ \quad b) $2^{3n} = 2^{12} \Rightarrow n = 4$ \quad c) $n = 8$ o $n = -8$
    \item $\dfrac{(40 - 40)^{5} + 36}{9} + (64 - 64)(25 + 10) = \dfrac{0 + 36}{9} + 0 \cdot 35 = 4$
  \end{enumerate}
