        \end{enumerate}

  \item \textbf{(5 pts)} \textbf{Signos, paréntesis y bases posibles:}
        \begin{enumerate}[label=\alph*)]
          \item Explique la diferencia entre $(-5)^2$ y $-5^2$, y compare ese caso con $(-2)^3$ y $-2^3$.
          \item Si $n^2 = 81$ en $\Z$, \textquestiondown qué valores puede tomar $n$? \textquestiondown Por qué en $n^3 = -27$ solo hay una solución entera?
        \end{enumerate}

  \item \textbf{(5 pts)} \textbf{Verdadero o Falso.} Indique si cada afirmación es verdadera (\textbf{V}) o falsa (\textbf{F}) y justifique con la regla correspondiente o un contraejemplo numérico:
        \begin{enumerate}[label=\alph*)]
          \item $(2 + 3)^2 = 2^2 + 3^2$.
          \item $2^3 + 2^4 = 2^7$.
          \item Si $x \neq 0$, entonces $5x^0 = 5$, pero $(5x)^0 = 1$.
          \item Para todo entero $a$, siempre se cumple que $2a = a^2$.
        \end{enumerate}

  \item \textbf{(5 pts)} \textbf{Propiedades y cambio de base:}
        \begin{enumerate}[label=\alph*)]
          \item Explique cómo distinguir el producto de potencias de igual base ($a^m \cdot a^n$) de la potencia de una potencia ($(a^m)^n$).
          \item Explique en qué consiste el «cambio de base» (descomposición en factores primos) y por qué permite simplificar $\dfrac{8^2}{4^3}$, pero no permite unir $9^2 \cdot 4^3$ en una sola potencia.
        \end{enumerate}
\end{enumerate}

  \subsection*{Parte Teórica (Preguntas 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item a) Una potencia abrevia una multiplicación repetida. En $4^{3} = 64$: base $= 4$, exponente $= 3$, potencia $= 64$. Es $4 \cdot 4 \cdot 4 = 64$, distinto de $4 \cdot 3 = 12$ (suma repetida).
          \quad b) $\dfrac{7^{2}}{7^{2}} = \dfrac{49}{49} = 1$ y, restando exponentes, $\dfrac{7^{2}}{7^{2}} = 7^{2-2} = 7^{0}$, luego $7^{0} = 1$. La expresión $0^{0}$ no está definida porque la justificación requiere dividir entre cero, lo cual no existe.

    \item a) $(-5)^{2} = (-5)(-5) = 25$ (el paréntesis incluye al signo) y $-5^{2} = -(5 \cdot 5) = -25$ (el signo queda afuera). En exponente impar coinciden numéricamente: $(-2)^{3} = -8$ y $-2^{3} = -8$, pero su estructura sigue siendo distinta.
          \quad b) $n^{2} = 81$ tiene dos soluciones: $n = 9$ y $n = -9$ (exponente par da positivo en ambos). En $n^{3} = -27$, solo $n = -3$, pues el exponente impar conserva el signo de la base.

    \item a) \textbf{F}: $(2+3)^{2} = 25 \neq 4 + 9 = 13$ (la potencia no se distribuye en la suma).
          \quad b) \textbf{F}: $2^{3} + 2^{4} = 8 + 16 = 24 \neq 2^{7} = 128$ (solo se suman exponentes al multiplicar igual base).
          \quad c) \textbf{V}: en $5x^{0}$ el $0$ afecta solo a $x$ ($5 \cdot 1 = 5$); en $(5x)^{0}$ afecta a todo el producto ($1$).
          \quad d) \textbf{F}: $2a = a + a$ y $a^{2} = a \cdot a$. Contraejemplo ($a = 5$): $2(5) = 10 \neq 5^{2} = 25$.

    \item a) En $a^{m} \cdot a^{n} = a^{m+n}$ hay dos potencias de igual base multiplicándose y se \textbf{suman} exponentes; en $(a^{m})^{n} = a^{m \cdot n}$ una potencia se eleva a otro exponente y se \textbf{multiplican} exponentes.
          \quad b) Reescribe bases compuestas como potencias de un mismo primo: $\dfrac{8^{2}}{4^{3}} = \dfrac{(2^{3})^{2}}{(2^{2})^{3}} = \dfrac{2^{6}}{2^{6}} = 1$. En cambio, $9^{2} \cdot 4^{3} = 3^{4} \cdot 2^{6}$ tiene bases primas distintas ($3$ y $2$) y exponentes distintos ($4$ y $6$), por lo que no se puede juntar en una sola potencia.
  \end{enumerate}

  \subsection*{Base, Signos y Operaciones Combinadas (Ejercicios 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item $(-2)^{4} = 16$; \ $-2^{4} = -16$; \ $(-3)^{3} = -27$; \ $-(-3)^{2} = -9$
    \item $(-1)^{2025} = -1$; $(-1)^{2026} = 1$; $0^{7} = 0$; $10^{0} = 1$; $-(-5)^{1} = 5$ \ $\Rightarrow$ $-1 + 1 + 0 + 1 + 5 = 6$
    \item a) $(-7)^{12} > 0$ (exponente par); \quad b) $(-8)^{9} < 0$ (exponente impar); \quad c) $(-5)^{0} = 1 > 0$; \quad d) $(-2)^{15} < 0$ (impar), pero el signo menos de afuera lo cambia a positivo: $-(-2)^{15} > 0$.
    \item $9 + \dfrac{36 \cdot 4}{16} - 3^{3} + 8 \cdot 6^{0} = 9 + 9 - 27 + 8 = -1$
  \end{enumerate}

  \subsection*{Propiedades de las Potencias y Cambio de Base (Ejercicios 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item a) $2^{6+4+1} = 2^{11}$ \quad b) $\dfrac{9^{8+3}}{9^{6}} = 9^{11-6} = 9^{5}$ \quad c) $2^{24} \div 2^{19} = 2^{5} = 32$
    \item a) $10^{3} = 1000$ \quad b) $10^{7} \div 10^{4} = 10^{3} = 1000$ \quad c) $\left(\dfrac{12}{4}\right)^{3} + \dfrac{15^{2}}{15^{2}} = 3^{3} + 1 = 28$
    \item a) $\dfrac{2^{10} \cdot 2^{6}}{2^{12}} = \dfrac{2^{16}}{2^{12}} = 2^{4} = 16$ \quad b) $(3^{6} \cdot 3^{6})^{3} = (3^{12})^{3} = 3^{36}$ \quad c) $(2^{18} \div 2^{14})^{4} = (2^{4})^{4} = 2^{16} = 65\,536$
    \item a) $\dfrac{(2^{2} \cdot 3)^{4} \cdot (2 \cdot 3^{2})^{2}}{(2 \cdot 3)^{4} \cdot 3^{4}} = \dfrac{2^{10} \cdot 3^{8}}{2^{4} \cdot 3^{8}} = 2^{6} = 64$
          \quad b) $\dfrac{(2 \cdot 5^{2})^{3} \cdot 2^{2}}{(2 \cdot 5)^{4} \cdot 5} = \dfrac{2^{5} \cdot 5^{6}}{2^{4} \cdot 5^{5}} = 2 \cdot 5 = 10$
  \end{enumerate}

  \subsection*{Potencias con Letras — Mini Álgebra (Ejercicios 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item a) $2(-3)^{2}(2)^{3} - 3(-3)^{2}(2)^{2} = 2(9)(8) - 3(9)(4) = 144 - 108 = 36$
          \quad b) $(-2)^{3} + 3(-2)^{2} - 5(-2) + 1 = -8 + 12 + 10 + 1 = 15$
    \item a) $4p^{6} \cdot 9p^{4} = 36p^{10}$ \quad b) $8x^{3} \cdot 3(1) = 24x^{3}$ \quad c) $(9x^{2} \cdot 4x^{6})^{3} = (36x^{8})^{3} = 46\,656\,x^{24}$
    \item a) $\left(a^{4}b^{3}\right)^{3} = a^{12}b^{9}$ \quad b) $\dfrac{9m^{4}n^{6} \cdot 8m^{3}n^{3}}{6m^{5}n^{6}} = \dfrac{72m^{7}n^{9}}{6m^{5}n^{6}} = 12m^{2}n^{3}$
    \item Arriba: $m^{6} \cdot n^{6}m^{2} \cdot n^{18} = m^{8}n^{24}$; abajo: $n^{4}m^{8}$; fracción: $n^{20}$; al cubo: $(n^{20})^{3} = n^{60}$.
  \end{enumerate}

  \subsection*{Problemas de Aplicación y Razonamiento (Ejercicios 1--4)}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item a) $2^{6} = 64$ bacterias \quad b) $2^{8} \div 2^{5} = 2^{3} = 8$ veces más bacterias
    \item a) Área $= x^{2} = 10^{2} = 100$ cm$^{2}$; Volumen $= y^{3} = 4^{3} = 64$ cm$^{3}$ \quad b) $(2x)^{2} = 4x^{2}$: el área se \textbf{cuadruplica}.
    \item a) $3^{n+2} = 3^{7} \Rightarrow n = 5$ \quad b) $2^{3n} = 2^{12} \Rightarrow n = 4$ \quad c) $n = 8$ o $n = -8$
    \item $\dfrac{(40 - 40)^{5} + 36}{9} + (64 - 64)(25 + 10) = \dfrac{0 + 36}{9} + 0 \cdot 35 = 4$
  \end{enumerate}
