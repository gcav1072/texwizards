\section{El Conjunto de los Números Enteros}
% ============================================================

\subsection{\textquestiondown Qué es el conjunto de los enteros?}

El conjunto de los \textbf{números enteros} (que se escribe $\Z$) está formado por todos los números que necesitamos para contar, para quedarnos sin nada\ldots\ y para estar \emph{por debajo de cero}:

\begin{cuadroteoria}
  \textbf{Definición:}
  \[
    \Z = \{ \dots, -3, -2, -1, 0, 1, 2, 3, \dots \}
  \]
  \begin{itemize}
    \item Los números \textbf{positivos} ($1, 2, 3, \dots$) están a la derecha del cero.
    \item Los números \textbf{negativos} ($-1, -2, -3, \dots$) están a la izquierda del cero.
    \item El \textbf{cero} ($0$) no es positivo ni negativo: es el punto de partida.
  \end{itemize}
\end{cuadroteoria}

\noindent Cada entero tiene un \textbf{opuesto}: el mismo número con el signo cambiado. El opuesto de $7$ es $-7$; el opuesto de $-12$ es $12$; y el opuesto de $0$ es $0$.

\noindent Podemos dibujarlos en la \textbf{recta numérica}, una recta donde cada número tiene su casita:

\begin{center}
  \begin{tikzpicture}[x=0.9cm]
    \draw[<->, thick] (-5.8,0) -- (5.8,0);
    \foreach \x in {-5,-4,...,5} {
        \draw (\x,0.15) -- (\x,-0.15) node[below] {$\x$};
      }
    \node[above] at (0,0.35) {\small $0$ es el punto de partida};
  \end{tikzpicture}
\end{center}

\noindent Caminar a la \textbf{derecha} es sumar; caminar a la \textbf{izquierda} es restar. Por ejemplo, para calcular $-3 + 5$: partimos de $-3$ y damos $5$ pasos a la derecha\ldots\ llegamos a $2$. Esta misma recta nos servirá, un poco más adelante, para \emph{comparar} números.

  \item \facil{} Escribe el opuesto de $-12$.

  \item \facil{} \textquestiondown Cuál es el único entero que no es positivo ni negativo?

\end{enumerate}

\section{Valor Absoluto}
% ============================================================

\subsection{La distancia al cero}

\begin{cuadroteoria}
  \textbf{Definición:} el \textbf{valor absoluto} de un número entero $a$ se escribe $|a|$ y es \emph{la distancia} desde $a$ hasta el cero en la recta numérica. La distancia siempre es positiva o cero.

  Ejemplos: $|5| = 5$, $|-5| = 5$, $|0| = 0$.
\end{cuadroteoria}

\noindent Los números $-5$ y $5$ están a la misma distancia del cero, por eso $|-5| = |5| = 5$.

\begin{cuadroadvertencia}
  \textbf{Ojo:} el valor absoluto \emph{nunca} es negativo. Las barras no «borran» el signo: solo miden cuánto se aleja el número del cero. Por eso $|-9| = 9$ (y no $-9$), y por eso $|a| = -4$ es imposible.
\end{cuadroadvertencia}

\noindent En esta guía el valor absoluto nos servirá para dos cosas: medir \emph{distancias} al cero y decidir \emph{cuál de dos negativos es mayor} (lo veremos en la sección siguiente).

\subsection{Ejercicios: Valor Absoluto}

  \item \facil{} $|-23|$ \ y \ $|23|$

\section{Relación de Orden en $\Z$}
% ============================================================

\subsection{Los símbolos de comparación}

Comparar dos números es decidir \emph{cuál es mayor}. Para eso usamos estos símbolos:

\begin{cuadroteoria}
  \begin{center}
    \renewcommand{\arraystretch}{1.25}
    \begin{tabular}{cl}
      \toprule
      \textbf{Símbolo} & \textbf{Se lee} \\
      \midrule
      $=$    & «igual a» \\
      $\neq$ & «distinto de» \\
      $<$    & «menor que» \\
      $>$    & «mayor que» \\
      $\leq$ & «menor o igual que» \\
      $\geq$ & «mayor o igual que» \\
      \bottomrule
    \end{tabular}
  \end{center}

  \smallskip
  \noindent \textbf{Truco para no confundirlos:} la \emph{boca abierta} siempre mira al número \textbf{mayor}. Así, $3 > 1$ se lee «$3$ es mayor que $1$», y $-7 < -3$ se lee «$-7$ es menor que $-3$».
\end{cuadroteoria}

\subsection{La regla de la recta numérica}

\begin{cuadroteoria}
  \textbf{Regla de oro del orden:}
  \[
    a < b \quad \Longleftrightarrow \quad a \text{ está a la \textbf{izquierda} de } b \text{ en la recta.}
  \]
  Por eso \textbf{avanzar hacia la derecha} siempre es \textbf{crecer}: cada número es mayor que todos los que quedan a su izquierda.
\end{cuadroteoria}

\begin{center}
  \begin{tikzpicture}[x=1.05cm]
    \draw[<->, thick] (-6.2,0) -- (6.2,0);
    \foreach \x in {-6,-5,...,6} {
        \draw (\x,0.14) -- (\x,-0.14);
        \node[below=2pt, font=\scriptsize] at (\x,0) {$\x$};
      }
    \node[above=6pt, font=\small] at (0,0) {$0$};
    \fill[mwnaranja!18] (-6.2,-1.02) rectangle (0,-0.78);
    \node[font=\scriptsize, text=mwrojonaranja] at (-3.1,-0.90) {NÚMEROS MENORES QUE $0$};
    \fill[mwprimario!12] (0,-1.02) rectangle (6.2,-0.78);
    \node[font=\scriptsize, text=mwprimario] at (3.1,-0.90) {NÚMEROS MAYORES QUE $0$};
  \end{tikzpicture}
\end{center}

\noindent De esta regla salen tres consecuencias que conviene memorizar:

\begin{cuadroextra}
  \begin{itemize}
    \item \textbf{Todo negativo es menor que cero} y \textbf{todo positivo es mayor que cero}:
          \[
            -100 < 0 < 1
          \]
    \item \textbf{Entre dos negativos es mayor el que está más cerca del cero} (el de \emph{menor} valor absoluto):
          \[
            -7 < -3 \qquad \text{porque } |-7| = 7 > 3 = |-3|.
          \]
    \item \textbf{Entre dos positivos es mayor el que está más lejos del cero} (el de \emph{mayor} valor absoluto):
          \[
            3 < 7 \qquad \text{porque } |3| = 3 < 7 = |7|.
          \]
  \end{itemize}
\end{cuadroextra}

\begin{cuadroadvertencia}
  \textbf{La trampa clásica:} en los negativos el orden del valor absoluto se \emph{invierte}. $9$ es mayor que $3$, pero $-9$ es \textbf{menor} que $-3$. Y en los contextos pasa lo mismo: una deuda de $15$ Bs. ($-15$) es \emph{peor} que una deuda de $8$ Bs. ($-8$), y $-12\,^{\circ}$C es \emph{más frío} que $-5\,^{\circ}$C.
\end{cuadroadvertencia}

\subsection{Propiedades del orden}

\begin{cuadroteoria}
  \textbf{Propiedades de la relación de orden en $\Z$:}
  \begin{itemize}
    \item \textbf{Tricotomía:} dados dos enteros $a$ y $b$, se cumple \emph{exactamente una} de estas tres posibilidades:
          \[
            a < b, \qquad a = b, \qquad a > b.
          \]
    \item \textbf{Transitiva:} si $a < b$ y $b < c$, entonces $a < c$.
          Ejemplo: $-8 < -3$ y $-3 < 2$, luego $-8 < 2$.
    \item \textbf{Monotonía de la suma:} si sumamos el \textbf{mismo} número a ambos lados de una desigualdad, esta se conserva:
          \[
            a < b \quad \Longrightarrow \quad a + c < b + c.
          \]
          Ejemplo: como $-5 < 2$, al sumar $-7$ en ambos lados queda $-12 < -5$.
    \item \textbf{Monotonía del producto:}
          \begin{itemize}
            \item Si multiplicamos por un número \textbf{positivo}, la desigualdad se conserva:
                  \[
                    a < b \ \text{ y }\ c > 0 \quad \Longrightarrow \quad a \cdot c < b \cdot c.
                  \]
            \item Si multiplicamos por un número \textbf{negativo}, la desigualdad \textbf{se invierte}:
                  \[
                    a < b \ \text{ y }\ c < 0 \quad \Longrightarrow \quad a \cdot c > b \cdot c.
                  \]
          \end{itemize}
          Ejemplo: $2 < 5$, y al multiplicar por $-3$ resulta $-6 > -15$.
  \end{itemize}
\end{cuadroteoria}

\subsection{Ejemplos en contexto}

\begin{enumerate}[leftmargin=*, label=\textbf{Ejemplo \arabic*.}]
  \item \textbf{(temperaturas)} En la madrugada el termómetro marcó $-6\,^{\circ}$C y al mediodía $5\,^{\circ}$C. Para saber cuál fue la temperatura menor comparamos:
        \[
          -6 < 5
        \]
        La madrugada fue más fría.

  \item \textbf{(deudas)} Mario debe $15$ Bs. ($-15$) y Ana debe $8$ Bs. ($-8$). \textquestiondown Quién está en peor situación? Como
        \[
          -15 < -8,
        \]
        Mario debe más.

  \item \textbf{(alturas)} Un ascensor está en el piso $5$ y otro en el sótano $-2$. Como $-2 < 5$, el segundo está más abajo.
\end{enumerate}

\section{Suma y Resta de Números Enteros}
% ============================================================

\noindent Ya sabemos ubicar, medir distancias y comparar enteros. Ahora vamos a \emph{operar} con ellos, empezando por la suma y la resta.


\subsection{Regla de los signos para sumar y restar}

\begin{cuadrosolucion}
  \textbf{Para sumas y restas:}
  \begin{itemize}
    \item \textbf{Igual signo:} suma los números y conserva el signo.
          \quad $-4 + (-3) = -7$ \quad y \quad $5 + 8 = 13$.
    \item \textbf{Signos diferentes:} resta el número con mayor valor absoluto menos el otro, y colocarás el signo del número más grande de los dos (con mayor valor absoluto)
          \quad $-7 + 4 = -3$ \quad y \quad $7 + (-4) = 3$.
  \end{itemize}
\end{cuadrosolucion}

%\begin{cuadrosolucion}
%  \textbf{Para restar:} restar un número es lo mismo que \textbf{sumar su opuesto}:
%  \[
%    a - b = a + (-b)
%  \]
%  Ejemplo: $5 - 9 = 5 + (-9) = -4$. \quad Ejemplo: $-3 - 4 = -3 + (-4) = -7$.
%\end{cuadrosolucion}

\begin{cuadroadvertencia}
  \textbf{Los signos dobles:} cuando aparecen dos signos juntos, los resolvemos así:
  \[
    + (+) = + \qquad +(-) = - \qquad -(+) = - \qquad -(-) = +
  \]
  Ejemplo: $8 - (-3) = 8 + 3 = 11$. \quad Ejemplo: $-5 + (-2) = -5 - 2 = -7$.
\end{cuadroadvertencia}

\noindent\textbf{Ejemplo resuelto:} La temperatura en la mañana era de $-4\,^{\circ}$C y subió $10\,^{\circ}$C. \textquestiondown Cuál es la nueva temperatura?

\[
  -4 + 10 = 6
\]

\noindent La nueva temperatura es $6\,^{\circ}$C.

\subsection{Propiedades de la suma en $\Z$}

\begin{cuadroteoria}
  \textbf{Propiedades de la adición de números enteros:}
  \begin{itemize}
    \item \textbf{Conmutativa:} el orden de los sumandos no altera la suma.
          \[
            a + b = b + a
          \]
          Ejemplo: $(-8) + (-7) = -15$ \ y \ $(-7) + (-8) = -15$.
    \item \textbf{Asociativa:} la forma de agrupar los sumandos no altera la suma.
          \[
            (a + b) + c = a + (b + c)
          \]
          Ejemplo: $[(-5) + (-3)] + 8 = (-8) + 8 = 0$ \ y \ $(-5) + [(-3) + 8] = (-5) + 5 = 0$.
    \item \textbf{Elemento neutro:} el $0$ es el neutro de la suma, porque no cambia el resultado.
          \[
            a + 0 = 0 + a = a
          \]
          Ejemplo: $(-3) + 0 = 0 + (-3) = -3$.
    \item \textbf{Elemento opuesto:} todo entero $a$ tiene un opuesto $-a$, y la suma de ambos es $0$.
          \[
            a + (-a) = (-a) + a = 0
          \]
          Ejemplo: $(-5) + 5 = 5 + (-5) = 0$.
  \end{itemize}
\end{cuadroteoria}

\noindent Estas propiedades son las que nos permiten \emph{reagrupar} y \emph{reordenar} los sumandos para calcular más rápido. Por ejemplo, en $-13 + 29 - 17 + 1$ conviene agrupar los negativos y los positivos:
\[
  (-13 - 17) + (29 + 1) = -30 + 30 = 0.
\]

\section{Multiplicación y División de Números Enteros}
% ============================================================

\subsection{La regla de los signos}

Multiplicar es sumar repetido. Por ejemplo, $3 \times (-2)$ significa \emph{tres veces perder $2$}: $(-2) + (-2) + (-2) = -6$. Y multiplicar dos negativos... \textquestiondown sabías que da positivo? Piensa en la doble negación: \emph{«no es cierto que no tengo deudas»} significa que sí tengo deudas... no, espera: \emph{«no es cierto que no gané»} significa que sí gané. ¡Dos negativos se anulan!

\begin{cuadroteoria}
  \textbf{Regla de los signos (multiplicación y división):}

  \begin{center}
    \renewcommand{\arraystretch}{1.3}
    \begin{tabular}{cc|cc}
      \toprule
      \textbf{1.\textsuperscript{er} signo} & \textbf{2.\textsuperscript{o} signo} & \textbf{Producto} & \textbf{Cociente} \\
      \midrule
      $+$ & $+$ & $+$ & $+$ \\
      $+$ & $-$ & $-$ & $-$ \\
      $-$ & $+$ & $-$ & $-$ \\
      $-$ & $-$ & $+$ & $+$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \noindent En palabras: \textbf{signos iguales} dan resultado \textbf{positivo}; \textbf{signos distintos} dan resultado \textbf{negativo}. Y con el truco de los amigos y enemigos:
  \begin{itemize}
    \item $+ \cdot + = +$ \quad y \quad $+ : + = +$ \quad (amigo de mi amigo, mi amigo)
    \item $+ \cdot - = -$ \quad y \quad $+ : - = -$ \quad (amigo de mi enemigo, mi enemigo)
    \item $- \cdot + = -$ \quad y \quad $- : + = -$ \quad (enemigo de mi amigo, mi enemigo)
    \item $- \cdot - = +$ \quad y \quad $- : - = +$ \quad (enemigo de mi enemigo, mi amigo)
  \end{itemize}
\end{cuadroteoria}

\noindent La misma regla vale para la \textbf{división}: $+ \div + = +$, $+ \div - = -$, $- \div + = -$ y $- \div - = +$.

\begin{cuadroadvertencia}
  \textbf{Dos avisos importantes:}
  \begin{itemize}
    \item Cero por cualquier número da cero: $0 \times (-7) = 0$.
    \item Cero dividido entre cualquier número (distinto de cero) da cero: $0 \div (-4) = 0$. Pero \textbf{no se puede dividir entre cero} (¡prohibido!).
  \end{itemize}
\end{cuadroadvertencia}

\noindent\textbf{Ejemplo resuelto 1:} $(-4) \times (-3) = 12$ (menos por menos da más).

\noindent\textbf{Ejemplo resuelto 2:} $(-20) \div (-4) = 5$ (menos entre menos da más).

\noindent\textbf{Ejemplo resuelto 3:} Un buceador desciende $3$ metros cada minuto. \textquestiondown A qué profundidad estará después de $4$ minutos? Si llamamos «bajar» a restar (negativo): $4 \times (-3) = -12$. Estará a $-12$ m (es decir, $12$ m bajo el agua).

\subsection{Propiedades de la multiplicación en $\Z$}

\begin{cuadroteoria}
  \textbf{Propiedades de la multiplicación de números enteros:}
  \begin{itemize}
    \item \textbf{Conmutativa:} el orden de los factores no altera el producto.
          \[
            a \cdot b = b \cdot a
          \]
          Ejemplo: $(-3) \cdot (-12) = 36$ \ y \ $(-12) \cdot (-3) = 36$.
    \item \textbf{Asociativa:} la forma de agrupar los factores no altera el producto.
          \[
            (a \cdot b) \cdot c = a \cdot (b \cdot c)
          \]
          Ejemplo: $[(-2) \cdot 5] \cdot (-3) = (-10) \cdot (-3) = 30$ \ y \
          $(-2) \cdot [5 \cdot (-3)] = (-2) \cdot (-15) = 30$.
    \item \textbf{Elemento neutro:} el $1$ es el neutro de la multiplicación.
          \[
            a \cdot 1 = 1 \cdot a = a
          \]
          Ejemplo: $(-30) \cdot 1 = 1 \cdot (-30) = -30$.
    \item \textbf{Factor cero:} todo número multiplicado por $0$ da $0$.
          \[
            a \cdot 0 = 0 \cdot a = 0
          \]
          Ejemplo: $45 \cdot 0 = 0 \cdot 45 = 0$.
    \item \textbf{Distributiva respecto a la adición y a la sustracción:}
          \[
            a \cdot (b + c) = a \cdot b + a \cdot c
            \qquad
            a \cdot (b - c) = a \cdot b - a \cdot c
          \]
          Ejemplo: $5 \cdot (7 - 3) = 5 \cdot 7 - 5 \cdot 3 = 35 - 15 = 20$.
  \end{itemize}
\end{cuadroteoria}

\noindent\textbf{Ejemplo resuelto (propiedad distributiva):}
\[
  -23 \cdot (-16 - 28) = (-23) \cdot (-16) - (-23) \cdot (-28) = 368 + 644 = 1012.
\]
Comprobamos: $-16 - 28 = -44$ y $-23 \cdot (-44) = 1012$. \ \textquestiondown Se cumple!

\noindent La propiedad distributiva también sirve para calcular mentalmente productos grandes:
\[
  15 \cdot 99 = 15 \cdot (100 - 1) = 1500 - 15 = 1485.
\]

\subsection{La jerarquía de las operaciones}

Cuando en una misma expresión aparecen sumas, restas, multiplicaciones y divisiones con números enteros, \textbf{no} operamos de izquierda a derecha sin importar el tipo de operación. Para resolver correctamente, debemos seguir la \textbf{jerarquía de las operaciones}:

\begin{cuadroteoria}
  \textbf{Reglas de jerarquía:}
  \begin{enumerate}
    \item \textbf{Sin signos de agrupación:}
          \begin{itemize}
            \item Primero se resuelven las \textbf{multiplicaciones y divisiones} de izquierda a derecha.
            \item Luego se resuelven las \textbf{sumas y restas} de izquierda a derecha.
          \end{itemize}
    \item \textbf{Con signos de agrupación:}
          \begin{itemize}
            \item Se resuelven las operaciones dentro de los símbolos de agrupación \textbf{de adentro hacia afuera}: primero paréntesis $(\ )$, luego corchetes $[\ ]$ y finalmente llaves $\{\ \}$.
            \item Dentro de cada nivel de agrupación, se respeta la jerarquía estándar (multiplicaciones/divisiones antes que sumas/restas).
          \end{itemize}
  \end{enumerate}
\end{cuadroteoria}

\noindent\textbf{Ejemplo resuelto 1 (Sin signos de agrupación):}
\[
  (-4) - (-6) : (-3)
\]
\begin{itemize}
  \item Primero realizamos la división: $(-6) : (-3) = 2$.
  \item Luego la resta: $(-4) - 2 = -6$.
\end{itemize}

\noindent\textbf{Ejemplo resuelto 2 (Con signos de agrupación):}
\[
  [6 - (-1) - (-13)] : (-5)
\]
\begin{itemize}
  \item Primero resolvemos el interior del corchete: $6 - (-1) - (-13) = 6 + 1 + 13 = 20$.
  \item Finalmente dividimos: $20 : (-5) = -4$.
\end{itemize}

\noindent\textbf{Ejemplo resuelto 3 (Multinivel con paréntesis y corchetes):}
\[
  18 : [6 - 3 \cdot (-4 : 2 + 1)] - 3
\]
\begin{itemize}
  \item Paréntesis interior (división primero): $-4 : 2 + 1 = -2 + 1 = -1$.
  \item Corchete (multiplicación primero): $6 - 3 \cdot (-1) = 6 - (-3) = 6 + 3 = 9$.
  \item División externa: $18 : 9 = 2$.
  \item Resta final: $2 - 3 = -1$.
\end{itemize}

\begin{cuadroadvertencia}
  \textbf{\textquestiondown Multiplicación o División primero?} Si aparecen consecutivamente sin paréntesis que alteren el orden, se resuelven estrictamente \textbf{de izquierda a derecha}. Ejemplo: $4 \cdot (-3) : (-2) : (-6) = (-12) : (-2) : (-6) = 6 : (-6) = -1$.
\end{cuadroadvertencia}
