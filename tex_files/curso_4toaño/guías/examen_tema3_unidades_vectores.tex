  \item \textbf{(4 pts)} Suma los vectores $\vec{a} = (6, -2)$ y $\vec{b} = (-2, 5)$.

          \item Escribe el vector resultante como par ordenado.
          \item Calcula el módulo de la resultante.

  \item \textbf{(6 pts)} Un vector $\vec{u}$ tiene módulo $12$ y forma un ángulo de $210^{\circ}$ con el eje $X$ positivo.
        \begin{enumerate}[label=\alph*)]
          \item Indica el cuadrante y el ángulo de referencia.
          \item Halla sus componentes $u_x$ y $u_y$.
          \item Verifica que $\sqrt{u_x^2 + u_y^2} = 12$.
        \end{enumerate}

  \item \textbf{(6 pts)} \textbf{De polares a cartesianas.} Convierte los siguientes puntos, indicando en qué cuadrante quedan:

  \item \textbf{(6 pts)} \textbf{De cartesianas a polares.} Convierte los siguientes puntos (ángulo exacto o con dos decimales, según corresponda):

          a)\ P(-1,\; \sqrt{3}) \qquad b)\ Q(-6,\; -8)

  \subsection*{Sección 2: Transformación de Unidades}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item a) $350 \div 100 = 3{,}5\;\text{m}$.
          \quad b) $4{,}5 \times 1\,000 = 4\,500\;\text{m}$.
          \quad c) $2\,500 \div 1\,000 = 2{,}5\;\text{kg}$.
          \quad d) $2 \times 3\,600 = 7\,200\;\text{s}$.
          \quad e) $540 \div 3{,}6 = 150\;\text{m/s}$.

    \item a) $15 \times 3{,}6 = 54\;\text{km/h}$. \quad b) $108 \div 3{,}6 = 30\;\text{m/s}$.

    \item $\rho = \dfrac{300}{40} = 7{,}5\;\text{g/cm}^3$. \quad En $\text{kg/m}^3$: $7{,}5 \times 1\,000 = 7\,500\;\text{kg/m}^3$.

    \item a) $42{,}195\;\text{km} = 42\,195\;\text{m}$.
          \quad b) $90 \div 3{,}6 = 25\;\text{m/s}$, así que recorre $25\;\text{m}$ en un segundo.
  \end{enumerate}

  \subsection*{Sección 3: Vectores y Coordenadas Polares}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item a) E \quad b) V \quad c) V \quad d) E

    \item Con $\cos 30^{\circ} = \dfrac{\sqrt{3}}{2}$ y $\sen 30^{\circ} = \dfrac{1}{2}$:
          \[
            v_x = 10\cos 30^{\circ} = 10 \cdot \frac{\sqrt{3}}{2} = 5\sqrt{3} \approx 8{,}66, \qquad
            v_y = 10\sen 30^{\circ} = 10 \cdot \frac{1}{2} = 5.
          \]
          Entonces $\vec{v} = (5\sqrt{3},\, 5)$.

    \item a) $\vec{a} + \vec{b} = (6 + (-2),\, -2 + 5) = (4, 3)$.
          \quad b) $|\vec{a} + \vec{b}| = \sqrt{4^2 + 3^2} = \sqrt{16 + 9} = \sqrt{25} = 5$.

    \item a) $210^{\circ}$ está en el \textbf{cuadrante III} y el ángulo de referencia es $\theta_{\text{ref}} = 210^{\circ} - 180^{\circ} = 30^{\circ}$.
          \quad b) Ambas componentes son negativas:
          \[
            u_x = -12\cos 30^{\circ} = -12 \cdot \frac{\sqrt{3}}{2} = -6\sqrt{3} \approx -10{,}39, \qquad
            u_y = -12\sen 30^{\circ} = -12 \cdot \frac{1}{2} = -6.
          \]
          \quad c) $\sqrt{(-6\sqrt{3})^2 + (-6)^2} = \sqrt{108 + 36} = \sqrt{144} = 12$. \ \checkmark

    \item a) $x = 10\cos 30^{\circ} = 10 \cdot \dfrac{\sqrt{3}}{2} = 5\sqrt{3} \approx 8{,}66$ y $y = 10\sen 30^{\circ} = 10 \cdot \dfrac{1}{2} = 5$, así que $P(5\sqrt{3},\, 5)$ (cuadrante I).
          \quad b) $240^{\circ}$ está en el cuadrante III con $\theta_{\text{ref}} = 60^{\circ}$:
          \[
            x = -20\cos 60^{\circ} = -10, \qquad y = -20\sen 60^{\circ} = -10\sqrt{3} \approx -17{,}32 \quad \Rightarrow \quad Q(-10,\, -10\sqrt{3}).
          \]
          Verificación: $\sqrt{(-10)^2 + (-10\sqrt{3})^2} = \sqrt{100 + 300} = \sqrt{400} = 20$. \ \checkmark

    \item a) $r = \sqrt{(-1)^2 + (\sqrt{3})^2} = \sqrt{4} = 2$; $\theta_{\text{ref}} = \arctan\dfrac{\sqrt{3}}{1} = 60^{\circ}$; como está en el cuadrante II, $\theta = 180^{\circ} - 60^{\circ} = 120^{\circ}$: $P(2,\, 120^{\circ})$.
          \quad b) $r = \sqrt{(-6)^2 + (-8)^2} = \sqrt{100} = 10$; $\theta_{\text{ref}} = \arctan\dfrac{8}{6} \approx 53{,}13^{\circ}$; como está en el cuadrante III, $\theta = 180^{\circ} + 53{,}13^{\circ} = 233{,}13^{\circ}$: $Q(10,\, 233{,}13^{\circ})$.
  \end{enumerate}

  \subsection*{Sección 4: Despeje de Fórmulas y Evaluación Numérica}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item $v\,t = d \Rightarrow t = \dfrac{d}{v} = \dfrac{300}{15} = 20\;\text{s}$.

    \item $\dfrac{9}{5}C = F - 32 \Rightarrow F = \dfrac{9}{5}C + 32 = \dfrac{9}{5}(25) + 32 = 45 + 32 = 77^{\circ}\text{F}$.

    \item $v_0^2 = v^2 - 2a\,\Delta x \Rightarrow v_0 = \sqrt{v^2 - 2a\,\Delta x} = \sqrt{25^2 - 2(3)(80)} = \sqrt{625 - 480} = \sqrt{145} \approx 12{,}0\;\text{m/s}$.

    \item a) $30 = 25t - 5t^2 \Rightarrow 5t^2 - 25t + 30 = 0 \Rightarrow t^2 - 5t + 6 = 0$. Factorizando: $(t - 2)(t - 3) = 0$, de donde $t = 2\;\text{s}$ y $t = 3\;\text{s}$.
          \quad b) Ambas soluciones son válidas porque son tiempos positivos: a los $2\;\text{s}$ el objeto \emph{sube} pasando por los $30\;\text{m}$ y a los $3\;\text{s}$ \emph{baja} pasando otra vez por esa altura.
  \end{enumerate}
