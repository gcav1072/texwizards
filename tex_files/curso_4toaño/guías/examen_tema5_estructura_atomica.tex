  \subsection*{Segunda Parte — Modelos Atómicos}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item \textbf{(A)} Dalton: el átomo es una esfera maciza e indivisible de materia.
    \item \textbf{(B)} Thomson: los electrones están incrustados en una masa de carga positiva («pan de pasas»).
    \item \textbf{(C)} Rutherford: el átomo tiene un núcleo pequeño, denso y positivo, con los electrones a su alrededor.
    \item \textbf{(D)} Bohr: los electrones se desplazan en órbitas definidas, cada una con un nivel de energía.
  \end{enumerate}

  \subsection*{Tercera Parte — Diagramas de Orbitales}

  \begin{enumerate}[leftmargin=*, label=\textbf{(\alph*)}]
    \item Orbital \textbf{$p$} orientado sobre el eje $x$: orbital $p_x$. (El color rojo representa la fase positiva de la función de onda y el azul la negativa).
    \item Orbital \textbf{$s$}: de forma esférica.
    \item Orbital \textbf{$p$} orientado sobre el eje $z$: orbital $p_z$.
    \item Orbital \textbf{$d$}: cuatro lóbulos (por ejemplo $d_{xy}$). El subnivel $d$ tiene \textbf{5 orbitales}.
    \item Orbital \textbf{$f$}: ocho lóbulos. El subnivel $f$ tiene 7 orbitales y aloja como máximo \textbf{14 electrones}.
  \end{enumerate}

  \subsection*{Cuarta Parte — Partículas Subatómicas}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item ${}^{31}_{15}\text{P}$: $Z = 15$, $A = 31$, $p = 15$, $n = 31 - 15 = 16$, $e = 15$ — \textbf{neutro}.
    \item ${}^{35}_{17}\text{Cl}$: $Z = 17$, $A = 35$, $p = 17$, $n = 35 - 17 = 18$, $e = 17$ — \textbf{neutro}.
    \item $\text{Mg}^{2+}$: $Z = 12$, $A = 24$, $p = 12$, $n = 24 - 12 = 12$, $e = 12 - 2 = 10$ — \textbf{catión}.
    \item $\text{O}^{2-}$: $Z = 8$, $A = 16$, $p = 8$, $n = 16 - 8 = 8$, $e = 8 + 2 = 10$ — \textbf{anión}.
    \item ${}^{40}_{20}\text{Ca}^{2+}$: $Z = 20$, $A = 40$, $p = 20$, $n = 40 - 20 = 20$, $e = 20 - 2 = 18$ — \textbf{catión}.
  \end{enumerate}

  \subsection*{Quinta Parte — Configuración Electrónica y Números Cuánticos}

  \begin{enumerate}[leftmargin=*, label=\textbf{\arabic*.}]
    \item a) Cl ($17$ e$^-$): $1s^{2}\,2s^{2}\,2p^{6}\,3s^{2}\,3p^{5}$.
          \quad b) Ca ($20$ e$^-$): $1s^{2}\,2s^{2}\,2p^{6}\,3s^{2}\,3p^{6}\,4s^{2}$.

    \item Azufre ($16$ e$^-$): $1s^{2}\,2s^{2}\,2p^{6}\,3s^{2}\,3p^{4}$ $\Rightarrow$ electrones de valencia $= 2 + 4 = 6$.

    \item Fósforo ($15$ e$^-$): $1s^{2}\,2s^{2}\,2p^{6}\,3s^{2}\,3p^{3}$.
          \\[2pt]
          Los tres electrones $3p$ se colocan \emph{sueltos} (regla de Hund), con espines paralelos:
          \[
            3p:\ \boxed{\uparrow}\ \boxed{\uparrow}\ \boxed{\uparrow}
          \]

    \item El azufre neutro tiene 16 electrones; el ion $\text{S}^{2-}$ ganó 2 electrones, así que tiene 18:
          \[
            \text{S}^{2-}:\ 1s^{2}\,2s^{2}\,2p^{6}\,3s^{2}\,3p^{6}.
          \]
          Esta es \textbf{exactamente} la configuración del argón ($Z = 18$): se dice que son \textbf{isoelectrónicos}.

    \item a) Configuración del fósforo: $1s^{2}\,2s^{2}\,2p^{6}\,3s^{2}\,3p^{3}$. El último subnivel es $3p$, con $n = 3$ y $l = 1$.
          Los tres electrones $3p$ van sueltos en los orbitales $-1, 0, +1$ con espín $+\tfrac{1}{2}$.
          Último electrón: $(n, l, m_l, m_s) = (3, 1, +1, +\tfrac{1}{2})$.
          \quad b) Configuración del calcio: $1s^{2}\,2s^{2}\,2p^{6}\,3s^{2}\,3p^{6}\,4s^{2}$.
          Último nivel: $4$ $\Rightarrow$ \textbf{periodo 4}. Electrones de valencia: $4s^{2}$ $\Rightarrow$ 2 electrones $\Rightarrow$ \textbf{grupo 2} (alcalinotérreos).
  \end{enumerate}
