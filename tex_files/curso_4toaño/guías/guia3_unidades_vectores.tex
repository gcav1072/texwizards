\section{Sistema Internacional de Unidades y medición}
% ============================================================

\subsection{Magnitudes y unidades}

\begin{cuadroteoria}
  Una \textbf{magnitud física} es toda propiedad que se puede medir (longitud, masa, tiempo, temperatura\ldots).
  \begin{itemize}[leftmargin=*]
    \item \textbf{Magnitudes fundamentales:} no se definen a partir de otras (longitud, masa, tiempo\ldots).
    \item \textbf{Magnitudes derivadas:} se obtienen combinando fundamentales (velocidad, fuerza, densidad\ldots).
  \end{itemize}
\end{cuadroteoria}
\newpage
\begin{cuadroteoria}
  \textbf{Unidades fundamentales del Sistema Internacional (SI):}
  \begin{center}
    \begin{tabular}{lll}
      \toprule
      Magnitud                & Unidad    & Símbolo \\
      \midrule
      Longitud                & metro     & m       \\
      Masa                    & kilogramo & kg      \\
      Tiempo                  & segundo   & s       \\
      Intensidad de corriente & ampere    & A       \\
      Temperatura             & kelvin    & K       \\
      Cantidad de sustancia   & mol       & mol     \\
      Intensidad luminosa     & candela   & cd      \\
      \bottomrule
    \end{tabular}
  \end{center}
\end{cuadroteoria}

\begin{cuadroteoria}
  \textbf{Prefijos del SI} (multiplicadores de las unidades):
  \begin{center}
    \begin{tabular}{llll}
      \toprule
      Prefijo & Símbolo & Factor    & Ejemplo                                     \\
      \midrule
      kilo    & k       & $10^{3}$  & $1 \text{ km} = 1000 \text{ m}$             \\
      hecto   & h       & $10^{2}$  & $1 \text{ hm} = 100 \text{ m}$              \\
      deca    & da      & $10^{1}$  & $1 \text{ dam} = 10 \text{ m}$              \\
      deci    & d       & $10^{-1}$ & $1 \text{ dm} = 0{,}1 \text{ m}$            \\
      centi   & c       & $10^{-2}$ & $1 \text{ cm} = 0{,}01 \text{ m}$           \\
      mili    & m       & $10^{-3}$ & $1 \text{ mm} = 0{,}001 \text{ m}$          \\
      micro   & $\mu$   & $10^{-6}$ & $1 \text{ }\mu\text{m} = 10^{-6} \text{ m}$ \\
      \bottomrule
    \end{tabular}
  \end{center}
\end{cuadroteoria}

\begin{cuadroteoria}
  \textbf{Notación científica:} se escribe un número como $a \times 10^{n}$, con $1 \le a < 10$ y $n$ entero.
  $$4\,500\,000 = 4{,}5 \times 10^{6}, \qquad 0{,}00025 = 2{,}5 \times 10^{-4}$$

  \textbf{Cifras significativas (introducción):} son los dígitos que aportan información real de una medición. Por ejemplo, $2{,}50$ m tiene tres cifras significativas (el cero cuenta).
\end{cuadroteoria}

\begin{cuadroadvertencia}
  \textbf{Errores clásicos:}
  \begin{itemize}[leftmargin=*]
    \item Escribir $0{,}0000001$ en lugar de $10^{-7}$: la notación científica evita errores de ceros.
    \item Confundir los prefijos: $1 \text{ cm} = 0{,}01 \text{ m}$ (centi es $\frac{1}{100}$), no $0{,}1$.
    \item Escribir unidades en plural o con mayúscula incorrecta: se escribe \emph{5 m} (no "5 ms", que es milisegundo) y \emph{10 kg} (no "10 Kgs").
  \end{itemize}
\end{cuadroadvertencia}

\subsection{Ejercicios resueltos}

\textbf{Ejemplo R1.} Clasifica cada magnitud en fundamental o derivada e indica su unidad del SI: $a)$ masa \quad $b)$ velocidad \quad $c)$ tiempo.

\begin{cuadrosolucion}
  \textbf{Solución:}
  \begin{itemize}
    \item $a)$ Masa: fundamental, se mide en kilogramos (kg).
    \item $b)$ Velocidad: derivada (longitud entre tiempo), se mide en m/s.
    \item $c)$ Tiempo: fundamental, se mide en segundos (s).
  \end{itemize}
\end{cuadrosolucion}

\textbf{Ejemplo R2.} Expresa en metros: $a)\ 5 \text{ km} \qquad b)\ 2500 \text{ mm}$.

\begin{cuadrosolucion}
  \textbf{Solución:}
  \begin{itemize}
    \item $a)\ 5 \text{ km} = 5 \times 10^{3} \text{ m} = 5000 \text{ m}$.
    \item $b)\ 2500 \text{ mm} = 2500 \times 10^{-3} \text{ m} = 2{,}5 \text{ m}$.
  \end{itemize}
\end{cuadrosolucion}

\textbf{Ejemplo R3.} Escribe en notación científica: $a)\ 0{,}00025 \qquad b)\ 4\,500\,000$.

\begin{cuadrosolucion}
  \textbf{Solución:}
  \begin{itemize}
    \item $a)\ 0{,}00025 = 2{,}5 \times 10^{-4}$ (la coma corre 4 lugares a la derecha).
    \item $b)\ 4\,500\,000 = 4{,}5 \times 10^{6}$ (la coma corre 6 lugares a la izquierda).
  \end{itemize}
\end{cuadrosolucion}

\section{Transformación de unidades}
% ============================================================

\subsection{El factor de conversión}

\begin{cuadroteoria}
  Un \textbf{factor de conversión} es una fracción que vale 1 y que relaciona dos unidades equivalentes. Por ejemplo:
  $$\frac{1000 \text{ m}}{1 \text{ km}} = 1 \qquad \text{y} \qquad \frac{1 \text{ km}}{1000 \text{ m}} = 1$$
  Al multiplicar una cantidad por un factor de conversión, la cantidad no cambia: solo cambia su forma de expresarla.
\end{cuadroteoria}

\begin{cuadropasos}
  \textbf{Cómo transformar unidades:}
  \begin{enumerate}[leftmargin=*]
    \item Escribe la cantidad con su unidad.
    \item Elige el factor de conversión que \emph{cancele} la unidad de partida (la unidad de partida debe quedar en el denominador del factor).
    \item Multiplica y simplifica: los números se multiplican y las unidades se cancelan.
  \end{enumerate}

  \textbf{Ejemplo:} convertir $150 \text{ cm}$ a metros:
  $$150 \text{ cm} \cdot \frac{1 \text{ m}}{100 \text{ cm}} = \frac{150}{100} \text{ m} = 1{,}5 \text{ m}$$
\end{cuadropasos}

\begin{cuadroteoria}
  \textbf{Conversión compuesta km/h $\leftrightarrow$ m/s:} como $1 \text{ km} = 1000 \text{ m}$ y $1 \text{ h} = 3600 \text{ s}$:
  $$1 \frac{\text{km}}{\text{h}} = \frac{1000 \text{ m}}{3600 \text{ s}} = \frac{1}{3{,}6} \frac{\text{m}}{\text{s}}$$
  \begin{itemize}[leftmargin=*]
    \item Para pasar de \textbf{km/h a m/s}: se divide entre $3{,}6$.
    \item Para pasar de \textbf{m/s a km/h}: se multiplica por $3{,}6$.
  \end{itemize}
\end{cuadroteoria}

\begin{cuadroteoria}
  \textbf{Densidad:} es la masa contenida en cada unidad de volumen.
  $$\rho = \frac{m}{V} \qquad (\text{en } \text{kg/m}^3 \text{ o } \text{g/cm}^3)$$
  El agua tiene densidad $1000 \text{ kg/m}^3 = 1 \text{ g/cm}^3$.
\end{cuadroteoria}

\begin{cuadroadvertencia}
  \textbf{Errores clásicos:}
  \begin{itemize}[leftmargin=*]
    \item Multiplicar en lugar de dividir: de km/h a m/s se \emph{divide} entre $3{,}6$ (72 km/h $= 20$ m/s, no $259{,}2$).
    \item Cancelar unidades que no corresponden: el factor debe colocarse de modo que la unidad de partida quede \emph{abajo}.
    \item Olvidar las unidades en el resultado final: un número sin unidad no es una medida física.
  \end{itemize}
\end{cuadroadvertencia}

\subsection{Ejercicios resueltos}

\textbf{Ejemplo R1.} Convierte $150 \text{ cm}$ a metros.

\begin{cuadrosolucion}
  \textbf{Solución:}
  $$150 \text{ cm} \cdot \frac{1 \text{ m}}{100 \text{ cm}} = 1{,}5 \text{ m}.$$
\end{cuadrosolucion}

\textbf{Ejemplo R2.} Convierte $2{,}5$ horas a segundos.

\begin{cuadrosolucion}
  \textbf{Solución:} usamos dos factores (horas a minutos y minutos a segundos):
  $$2{,}5 \text{ h} \cdot \frac{60 \text{ min}}{1 \text{ h}} \cdot \frac{60 \text{ s}}{1 \text{ min}} = 2{,}5 \cdot 60 \cdot 60 \text{ s} = 9000 \text{ s}.$$
\end{cuadrosolucion}

\textbf{Ejemplo R3.} Un autobús va a $72$ km/h. ¿A cuántos m/s equivale?

\begin{cuadrosolucion}
  \textbf{Solución:} de km/h a m/s se divide entre $3{,}6$:
  $$72 \frac{\text{km}}{\text{h}} \div 3{,}6 = 20 \frac{\text{m}}{\text{s}}.$$
\end{cuadrosolucion}

\textbf{Ejemplo R4.} Un bloque de $2 \text{ kg}$ ocupa un volumen de $0{,}004 \text{ m}^3$. Calcula su densidad.

\begin{cuadrosolucion}
  \textbf{Solución:}
  $$\rho = \frac{m}{V} = \frac{2 \text{ kg}}{0{,}004 \text{ m}^3} = 500 \frac{\text{kg}}{\text{m}^3}.$$
\end{cuadrosolucion}

\section{Vectores}
% ============================================================

\subsection{Escalares y vectores}

\begin{cuadroteoria}
  \begin{itemize}[leftmargin=*]
    \item \textbf{Magnitud escalar:} queda definida solo con un número y su unidad. Ejemplos: masa ($5$ kg), tiempo ($10$ s), temperatura ($30^{\circ}$C).
    \item \textbf{Magnitud vectorial:} necesita módulo, dirección y sentido. Ejemplos: desplazamiento, velocidad, fuerza.
  \end{itemize}

  Un \textbf{vector} se representa con una flecha y se denota $\vec{v}$:
  \begin{center}
    \begin{tikzpicture}[scale=1]
      \draw[thick, -Latex, mwprimario] (0,0) -- (3.5,1.75);
      \filldraw (0,0) circle (2pt) node[below left] {origen};
      \node[mwprimario] at (3.9,1.9) {$\vec{v}$};
      \draw[mwrojonaranja, thick, -Latex] (4.5,0) -- (4.5,2) node[right] {módulo (longitud)};
      \draw[gray, dashed] (0,0) -- (3.5,0);
      \draw[mwvioleta, thick] (0.7,0) arc (0:26.6:0.7);
      \node[mwvioleta] at (1.05,0.25) {$\theta$};
    \end{tikzpicture}
  \end{center}
  \begin{itemize}[leftmargin=*]
    \item \textbf{Módulo} $|\vec{v}|$: la longitud de la flecha (la "cantidad" del vector).
    \item \textbf{Dirección}: la recta sobre la que actúa (dada por el ángulo $\theta$).
    \item \textbf{Sentido}: hacia dónde apunta la flecha.
  \end{itemize}
\end{cuadroteoria}

\subsection{Componentes rectangulares}

\begin{cuadroteoria}
  Todo vector se puede descomponer en dos \textbf{componentes rectangulares} (una sobre el eje $X$ y otra sobre el eje $Y$):
  $$\vec{v} = (v_x, v_y) \qquad \text{con} \qquad v_x = |\vec{v}| \cos \theta, \quad v_y = |\vec{v}| \sen \theta$$
  \begin{center}
    \begin{tikzpicture}[scale=0.95]
      \draw[thin, gray!40] (-0.5,-0.5) grid (4.5,3.5);
      \draw[thick, -Latex] (-0.8,0) -- (4.8,0) node[right] {$x$};
      \draw[thick, -Latex] (0,-0.8) -- (0,3.8) node[above] {$y$};
      \draw[thick, -Latex, mwprimario] (0,0) -- (3.6,2.4) node[above right] {$\vec{v}$};
      \draw[mwprimario, thick] (0.7,0) arc (0:33.7:0.7);
      \node[mwprimario] at (1.0,0.25) {$\theta$};
      \draw[mwmagenta, thick, -Latex] (0,0) -- (3.6,0) node[midway, below] {$v_x$};
      \draw[mwmagenta, thick, -Latex] (3.6,0) -- (3.6,2.4) node[midway, right] {$v_y$};
      \draw[gray, dashed] (0,0) -- (0,2.4);
      \draw[gray, dashed] (3.6,2.4) -- (0,2.4);
    \end{tikzpicture}
  \end{center}
  Y al revés, conociendo las componentes se puede recuperar el módulo con Pitágoras:
  $$|\vec{v}| = \sqrt{v_x^2 + v_y^2}$$
\end{cuadroteoria}

\subsection{Suma de vectores}

\begin{cuadroteoria}
  \textbf{Método del polígono (gráfico):} se dibujan los vectores uno seguido de otro (el origen de cada uno en la punta del anterior). El vector resultante une el origen del primero con la punta del último.
  \begin{center}
    \begin{tikzpicture}[scale=0.9]
      \draw[thick, -Latex, mwprimario] (0,0) -- (2.5,1.2) node[midway, above] {$\vec{a}$};
      \draw[thick, -Latex, mwnaranja] (2.5,1.2) -- (4.5,0.4) node[midway, above] {$\vec{b}$};
      \draw[thick, -Latex, mwmagenta] (0,0) -- (4.5,0.4) node[midway, below] {$\vec{a} + \vec{b}$};
    \end{tikzpicture}
  \end{center}

  \textbf{Método analítico (por componentes):} se suman las componentes de cada eje por separado.
  $$\vec{a} = (a_x, a_y), \quad \vec{b} = (b_x, b_y) \quad \Rightarrow \quad \vec{a} + \vec{b} = (a_x + b_x, a_y + b_y)$$

  \textbf{Multiplicar un vector por un escalar $k$:} cambia el módulo (se alarga o acorta) y, si $k < 0$, invierte el sentido.
\end{cuadroteoria}

\begin{cuadropasos}
  \textbf{Sumar vectores por componentes:}
  \begin{enumerate}[leftmargin=*]
    \item Descompón cada vector en sus componentes con $\cos \theta$ y $\sen \theta$.
    \item Suma las componentes $x$ entre sí y las $y$ entre sí.
    \item Calcula el módulo de la resultante con Pitágoras.
  \end{enumerate}
\end{cuadropasos}

\begin{cuadroadvertencia}
  \textbf{Errores clásicos:}
  \begin{itemize}[leftmargin=*]
    \item Sumar vectores como si fueran números: $3$ km al este + $4$ km al norte NO son $7$ km: son $\sqrt{3^2 + 4^2} = 5$ km.
    \item Confundir dirección con sentido: "norte" es una dirección, "hacia el norte" es un sentido.
    \item Olvidar el signo de las componentes: un vector que apunta a la izquierda o hacia abajo tiene componentes negativas.
  \end{itemize}
\end{cuadroadvertencia}

\subsection{Reducción al primer cuadrante}

\begin{cuadroteoria}
  Cuando un vector forma un ángulo $\theta$ \emph{arbitrario} con el eje $X$ positivo, se habla de su \textbf{ángulo de referencia} $\theta_{\text{ref}}$ (siempre entre $0^{\circ}$ y $90^{\circ}$) y del \textbf{cuadrante} en que se ubica su punta.  El ángulo de referencia es el ángulo agudo formado con el eje $X$ (positivo o negativo, según el cuadrante).

  \begin{center}
    \begin{tikzpicture}[scale=1.05]
      % ejes
      \draw[thick, -Latex] (-3,0) -- (3,0) node[right] {$x$};
      \draw[thick, -Latex] (0,-3) -- (0,3) node[above] {$y$};
      % etiquetas de cuadrantes
      \node[mwprimario!80, font=\small\bfseries] at ( 1.6, 1.6) {I};
      \node[mwnaranja!90, font=\small\bfseries]  at (-1.6, 1.6) {II};
      \node[mwmagenta!90, font=\small\bfseries]  at (-1.6,-1.6) {III};
      \node[mwvioleta!90, font=\small\bfseries]  at ( 1.6,-1.6) {IV};
      % vectores ejemplo en cada cuadrante
      \draw[thick,-Latex,mwprimario]  (0,0) -- ( 1.8, 1.2) node[above right,font=\tiny] {$\vec{v}_{\text{I}}$};
      \draw[thick,-Latex,mwnaranja]   (0,0) -- (-1.8, 1.2) node[above left, font=\tiny] {$\vec{v}_{\text{II}}$};
      \draw[thick,-Latex,mwmagenta]   (0,0) -- (-1.8,-1.2) node[below left, font=\tiny] {$\vec{v}_{\text{III}}$};
      \draw[thick,-Latex,mwvioleta]   (0,0) -- ( 1.8,-1.2) node[below right,font=\tiny] {$\vec{v}_{\text{IV}}$};
    \end{tikzpicture}
  \end{center}
\end{cuadroteoria}
\newpage
\begin{cuadroteoria}
  \textbf{Signos de las componentes según el cuadrante:}
  \begin{center}
    \begin{tabular}{cccc}
      \toprule
      Cuadrante & $\theta$ (eje $+x$)              & $v_x$ & $v_y$ \\
      \midrule
      I         & $0^\circ < \theta < 90^\circ$    & $+$   & $+$   \\
      II        & $90^\circ < \theta < 180^\circ$  & $-$   & $+$   \\
      III       & $180^\circ < \theta < 270^\circ$ & $-$   & $-$   \\
      IV        & $270^\circ < \theta < 360^\circ$ & $+$   & $-$   \\
      \bottomrule
    \end{tabular}
  \end{center}

  Las fórmulas de las componentes \textbf{siempre se aplican con el ángulo de referencia $\theta_{\text{ref}}$}; el signo se asigna según el cuadrante:
  $$v_x = \pm |\vec{v}|\cos\theta_{\text{ref}}, \qquad v_y = \pm |\vec{v}|\sen\theta_{\text{ref}}$$
\end{cuadroteoria}

\begin{cuadropasos}
  \textbf{Cómo descomponer un vector en cualquier cuadrante:}
  \begin{enumerate}[leftmargin=*]
    \item Localiza el cuadrante en que está el vector a partir de $\theta$ (medido desde el eje $+x$ en sentido antihorario).
    \item Calcula el ángulo de referencia $\theta_{\text{ref}}$:
          \begin{itemize}
            \item Cuadrante I:   $\theta_{\text{ref}} = \theta$
            \item Cuadrante II:  $\theta_{\text{ref}} = 180^\circ - \theta$
            \item Cuadrante III: $\theta_{\text{ref}} = \theta - 180^\circ$
            \item Cuadrante IV:  $\theta_{\text{ref}} = 360^\circ - \theta$
          \end{itemize}
    \item Aplica $v_x = |\vec{v}|\cos\theta_{\text{ref}}$ y $v_y = |\vec{v}|\sen\theta_{\text{ref}}$.
    \item Asigna el signo correcto según el cuadrante (tabla de arriba).
  \end{enumerate}
\end{cuadropasos}

\begin{cuadroadvertencia}
  \textbf{Errores clásicos en la reducción al primer cuadrante:}
  \begin{itemize}[leftmargin=*]
    \item Usar $\theta$ directamente en $\cos$ y $\sen$ sin reducir: $\cos 150^\circ \neq \cos 150^\circ$ como positivo; hay que reconocer que da $-\cos 30^\circ$.
    \item Olvidar el signo de la componente: un vector en el III cuadrante tiene \emph{ambas} componentes negativas.
    \item Confundir el ángulo de referencia con el ángulo desde el eje $+y$: el ángulo de referencia siempre se mide desde el eje $X$ más cercano.
  \end{itemize}
\end{cuadroadvertencia}

\subsection{Coordenadas polares y cartesianas}

\begin{cuadroteoria}
  Un mismo punto (o la punta de un vector) se puede ubicar de dos maneras:
  \begin{itemize}[leftmargin=*]
    \item \textbf{Coordenadas cartesianas} $(x, y)$: cuánto me muevo horizontalmente y cuánto verticalmente.
    \item \textbf{Coordenadas polares} $(r, \theta)$: a qué \textbf{distancia} $r$ estoy del origen y con qué \textbf{ángulo} $\theta$ (medido desde el eje $+x$, en sentido antihorario).
  \end{itemize}
  \begin{center}
    \begin{tikzpicture}[scale=0.95]
      \draw[thin, gray!40] (-0.5,-0.5) grid (4.5,3.5);
      \draw[thick, -Latex] (-0.8,0) -- (4.8,0) node[right] {$x$};
      \draw[thick, -Latex] (0,-0.8) -- (0,3.8) node[above] {$y$};
      \draw[thick, -Latex, mwprimario] (0,0) -- (3.6,2.4) node[above right] {$P(r,\theta)$};
      \draw[mwprimario, thick] (0.7,0) arc (0:33.7:0.7);
      \node[mwprimario] at (1.0,0.25) {$\theta$};
      \node[mwprimario] at (1.6,1.45) {$r$};
      \draw[gray, dashed] (3.6,2.4) -- (3.6,0) node[below right, black] {$x$};
      \draw[gray, dashed] (3.6,2.4) -- (0,2.4) node[left, black] {$y$};
    \end{tikzpicture}
  \end{center}

  \textbf{De polares a cartesianas:} son las mismas fórmulas de las componentes rectangulares.
  $$x = r\cos\theta, \qquad y = r\sen\theta$$

  \textbf{De cartesianas a polares:} el módulo sale de Pitágoras y el ángulo, de la tangente.
  $$r = \sqrt{x^2 + y^2}, \qquad \tan\theta_{\text{ref}} = \left|\frac{y}{x}\right|$$
\end{cuadroteoria}

\begin{cuadroteoria}
  \textbf{El ángulo hay que llevarlo al cuadrante correcto} (la calculadora solo devuelve valores entre $-90^{\circ}$ y $90^{\circ}$):
  \begin{center}
    \begin{tabular}{lll}
      \toprule
      Cuadrante & Signos de $(x, y)$ & Ángulo real $\theta$ (con $\theta_{\text{ref}}$ agudo) \\
      \midrule
      I   & $x > 0,\ y > 0$ & $\theta = \theta_{\text{ref}}$              \\
      II  & $x < 0,\ y > 0$ & $\theta = 180^{\circ} - \theta_{\text{ref}}$ \\
      III & $x < 0,\ y < 0$ & $\theta = 180^{\circ} + \theta_{\text{ref}}$ \\
      IV  & $x > 0,\ y < 0$ & $\theta = 360^{\circ} - \theta_{\text{ref}}$ \\
      \bottomrule
    \end{tabular}
  \end{center}
  \textbf{Atajo:} dibuja el punto en el plano; así sabes de inmediato en qué cuadrante cae y cuál fórmula usar.
\end{cuadroteoria}

\begin{cuadropasos}
  \textbf{Cómo convertir coordenadas:}
  \begin{enumerate}[leftmargin=*]
    \item \textbf{Polares $\to$ cartesianas:} sustituye en $x = r\cos\theta$ y $y = r\sen\theta$ (con el ángulo \emph{completo}, no el de referencia).
    \item \textbf{Cartesianas $\to$ polares:} calcula $r = \sqrt{x^2 + y^2}$.
    \item Halla el ángulo de referencia con $\tan\theta_{\text{ref}} = \left|\dfrac{y}{x}\right|$ (usa la calculadora y redondea según lo pida el ejercicio).
    \item Ubica el cuadrante según los signos de $x$ y $y$ y ajusta el ángulo con la tabla.
    \item Verifica: al volver a polares $\to$ cartesianas debes recuperar las componentes originales.
  \end{enumerate}
\end{cuadropasos}

\begin{cuadroadvertencia}
  \textbf{Errores clásicos en los cambios de coordenadas:}
  \begin{itemize}[leftmargin=*]
    \item Dejar $\theta$ como lo da la calculadora: $\arctan(-1) = -45^{\circ}$ no describe un punto del cuadrante IV con la convención de $0^{\circ}$ a $360^{\circ}$, que corresponde a $315^{\circ}$.
    \item Escribir $r$ negativo: por convención $r \ge 0$; el signo de las componentes ya lo aportan $\cos\theta$ y $\sen\theta$.
    \item Confundir grados con radianes: la calculadora debe estar en el modo correcto (aquí trabajamos en grados).
    \item Usar el ángulo de referencia para hallar $r$: $r$ depende de $x$ e $y$, no del ángulo.
    \item Mezclar $(x,y)$ con $(r,\theta)$ en la misma respuesta: siempre indica qué sistema estás usando.
  \end{itemize}
\end{cuadroadvertencia}

\subsection{Ejercicios resueltos}

\textbf{Ejemplo R1.} Clasifica cada magnitud como escalar o vectorial: $a)$ $25^{\circ}$C \quad $b)$ $60$ km/h hacia el este \quad $c)$ $5$ kg \quad $d)$ fuerza de $10$ N hacia arriba.

\begin{cuadrosolucion}
  \textbf{Solución:}
  \begin{itemize}
    \item $a)$ Temperatura: escalar.
    \item $b)$ Velocidad con dirección: vectorial.
    \item $c)$ Masa: escalar.
    \item $d)$ Fuerza con dirección: vectorial.
  \end{itemize}
\end{cuadrosolucion}

\textbf{Ejemplo R2.} Un vector $\vec{v}$ tiene módulo $10$ y forma un ángulo de $60^{\circ}$ con el eje $X$ positivo. Halla sus componentes.

\begin{cuadrosolucion}
  \textbf{Solución:} usando los valores notables ($\cos 60^{\circ} = \tfrac{1}{2}$, $\sen 60^{\circ} = \tfrac{\sqrt{3}}{2}$):
  $$v_x = 10 \cos 60^{\circ} = 10 \cdot \frac{1}{2} = 5, \qquad v_y = 10 \sen 60^{\circ} = 10 \cdot \frac{\sqrt{3}}{2} = 5\sqrt{3}.$$
  Entonces $\vec{v} = (5,\, 5\sqrt{3})$.
\end{cuadrosolucion}

\textbf{Ejemplo R3.} Suma $\vec{a} = (3, 4)$ y $\vec{b} = (1, 2)$ y calcula el módulo de la resultante.

\begin{cuadrosolucion}
  \textbf{Solución:}
  $$\vec{a} + \vec{b} = (3 + 1, 4 + 2) = (4, 6), \qquad |\vec{a} + \vec{b}| = \sqrt{4^2 + 6^2} = \sqrt{16 + 36} = \sqrt{52} = 2\sqrt{13}.$$
\end{cuadrosolucion}

\textbf{Ejemplo R4.} Un vector $\vec{u}$ tiene módulo $10$ y forma un ángulo de $150^{\circ}$ con el eje $X$ positivo. Halla sus componentes.

\begin{cuadrosolucion}
  \textbf{Solución:} $150^{\circ}$ está en el \textbf{cuadrante II}, así que $v_x < 0$ y $v_y > 0$.
  \begin{enumerate}[leftmargin=*, label=\arabic*.]
    \item Ángulo de referencia: $\theta_{\text{ref}} = 180^{\circ} - 150^{\circ} = 30^{\circ}$.
    \item Componentes (con valores notables $\cos 30^{\circ} = \frac{\sqrt{3}}{2}$, $\sen 30^{\circ} = \frac{1}{2}$):
          $$v_x = -\,10\cos 30^{\circ} = -\,10 \cdot \frac{\sqrt{3}}{2} = -5\sqrt{3} \approx -8{,}66,$$
          $$v_y = +\,10\sen 30^{\circ} = +\,10 \cdot \frac{1}{2} = 5.$$
  \end{enumerate}
  Resultado: $\vec{u} = (-5\sqrt{3},\; 5)$.
\end{cuadrosolucion}

\textbf{Ejemplo R5.} Un vector $\vec{w}$ tiene módulo $8$ y apunta en la dirección $225^{\circ}$ (cuadrante III). Halla sus componentes.

\begin{cuadrosolucion}
  \textbf{Solución:} $225^{\circ}$ está en el \textbf{cuadrante III}: ambas componentes son negativas.
  \begin{enumerate}[leftmargin=*, label=\arabic*.]
    \item Ángulo de referencia: $\theta_{\text{ref}} = 225^{\circ} - 180^{\circ} = 45^{\circ}$.
    \item Con $\cos 45^{\circ} = \sen 45^{\circ} = \frac{\sqrt{2}}{2}$:
          $$w_x = -\,8\cos 45^{\circ} = -4\sqrt{2} \approx -5{,}66, \qquad w_y = -\,8\sen 45^{\circ} = -4\sqrt{2} \approx -5{,}66.$$
  \end{enumerate}
  Resultado: $\vec{w} = (-4\sqrt{2},\; -4\sqrt{2})$.
\end{cuadrosolucion}

\textbf{Ejemplo R6.} Convierte a coordenadas cartesianas el punto $P(8,\; 120^{\circ})$.

\begin{cuadrosolucion}
  \textbf{Solución:} usamos el ángulo completo ($120^{\circ}$ está en el cuadrante II, así que $x < 0$ e $y > 0$):
  $$x = 8\cos 120^{\circ} = -\,8 \cdot \frac{1}{2} = -4, \qquad y = 8\sen 120^{\circ} = 8 \cdot \frac{\sqrt{3}}{2} = 4\sqrt{3} \approx 6{,}93.$$
  Resultado: $P(-4,\; 4\sqrt{3})$; es decir, en coordenadas cartesianas $(-4;\ 6{,}93)$.
\end{cuadrosolucion}

\textbf{Ejemplo R7.} Convierte a coordenadas polares el punto $P(-3,\; 3)$ (ángulo exacto).

\begin{cuadrosolucion}
  \textbf{Solución:}
  \begin{enumerate}[leftmargin=*, label=\arabic*.]
    \item Módulo: $r = \sqrt{(-3)^2 + 3^2} = \sqrt{18} = 3\sqrt{2} \approx 4{,}24$.
    \item Ángulo de referencia: $\tan\theta_{\text{ref}} = \left|\dfrac{3}{-3}\right| = 1 \Rightarrow \theta_{\text{ref}} = 45^{\circ}$.
    \item El punto está en el \textbf{cuadrante II} ($x < 0$, $y > 0$), así que $\theta = 180^{\circ} - 45^{\circ} = 135^{\circ}$.
  \end{enumerate}
  Resultado: $P(3\sqrt{2},\; 135^{\circ})$. \textbf{Verificación:} $3\sqrt{2}\cos 135^{\circ} = 3\sqrt{2}\left(-\frac{\sqrt{2}}{2}\right) = -3$ y $3\sqrt{2}\sen 135^{\circ} = 3$. \checkmark
\end{cuadrosolucion}

\section{Despeje de fórmulas y evaluación numérica}

\subsection{¿Qué significa despejar una fórmula?}

\begin{cuadroteoria}
  \textbf{Despejar} es dejar \emph{sola} la variable que nos interesa a un lado de la igualdad. No buscamos un número: buscamos una \textbf{nueva fórmula} que nos diga cómo calcular esa variable usando las demás.

  \textbf{Ejemplo:} de la densidad $\rho = \dfrac{m}{V}$ salen tres versiones equivalentes de la misma relación:
  $$\rho = \frac{m}{V}, \qquad m = \rho \cdot V, \qquad V = \frac{m}{\rho}$$
\end{cuadroteoria}

\begin{cuadroteoria}
  \textbf{Regla de oro: aplica la operación inversa.} Lo que acompaña a la variable debe ``pasar al otro lado'' haciendo lo contrario:
  \begin{center}
    \begin{tabular}{ll}
      \toprule
      Si la variable está\ldots        & Aplica\ldots                                          \\
      \midrule
      Sumando o restando              & la operación inversa (resta o suma)                    \\
      Multiplicando                   & divide entre ese factor                                \\
      En el numerador (dividiendo)    & multiplica por el denominador                          \\
      En el denominador               & multiplica al otro lado                                \\
      Elevada al cuadrado             & raíz cuadrada                                          \\
      Dentro de una raíz cuadrada     & eleva al cuadrado                                      \\
      Elevada a la $n$                & raíz $n$-ésima ($\sqrt[n]{\phantom{x}}$)               \\
      \bottomrule
    \end{tabular}
  \end{center}
\end{cuadroteoria}

\begin{cuadropasos}
  \textbf{Procedimiento sugerido (de afuera hacia adentro):}
  \begin{enumerate}[leftmargin=*]
    \item Identifica la variable que vas a despejar y fíjate \emph{dónde} aparece (una sola vez, en un denominador, dentro de una raíz\ldots).
    \item Elimina primero lo que está \emph{más lejos} de ella: sumas y restas.
    \item Sigue con lo que está \emph{más cerca}: multiplicaciones y divisiones.
    \item Al final resuelve lo ``difícil'': raíces y potencias.
    \item Verifica: sustituye la fórmula despejada en la original y comprueba que se cumple la igualdad.
  \end{enumerate}
\end{cuadropasos}

\begin{cuadroextra}
  \textbf{Caso especial: la variable aparece dos veces.} Cuando la incógnita está en dos términos de la fórmula, primero se \emph{reúnen} sus términos y luego se \textbf{factoriza}. Por ejemplo, en $m = \rho_1 V_1 + \rho_2 V_2$ con $V_2 = V - V_1$ aparece $V_1$ dos veces: hay que escribir $m = (\rho_1 - \rho_2)V_1 + \rho_2 V$ antes de despejar. No hay que memorizar nada nuevo: es el mismo despeje con un paso extra.
\end{cuadroextra}

\subsection{Del despeje al valor numérico}

\begin{cuadroteoria}
  Una vez despejada la variable, \textbf{evaluar} (o hallar el valor numérico) consiste en \textbf{sustituir} los valores conocidos ---¡con sus unidades!--- y hacer las cuentas.

  \textbf{Ejemplo:} si $d = v \cdot t$, con $v = 20 \text{ m/s}$ y $t = 3 \text{ s}$:
  $$d = 20 \frac{\text{m}}{\text{s}} \cdot 3 \text{ s} = 60 \text{ m}$$
  Las unidades se multiplican igual que los números: $\frac{\text{m}}{\text{s}} \cdot \text{s} = \text{m}$.
\end{cuadroteoria}

\begin{cuadroadvertencia}
  \textbf{Errores clásicos al despejar y evaluar:}
  \begin{itemize}[leftmargin=*]
    \item ``Pasar'' un factor que multiplica a \emph{toda una suma}: en $P = 2\ell + 2w$ no se pasa el $2$ primero; se resta $2\ell$ y después se divide entre $2$.
    \item Despejar otra variable de la que pide el ejercicio.
    \item Cambiar el significado de una fórmula: despejar no es mover números ``al azar'', es aplicar la misma operación en \emph{ambos} lados.
    \item Olvidar las unidades al sustituir: $\rho$ en kg/m$^3$, $t$ en segundos, $v$ en m/s. Si las mezclas, el resultado pierde sentido.
    \item Escoger la raíz negativa cuando el resultado es una longitud, un tiempo o una rapidez: esas magnitudes son positivas.
  \end{itemize}
\end{cuadroadvertencia}

\subsection{Cuando el despeje conlleva una ecuación de segundo grado}

\begin{cuadroteoria}
  Si la variable que quieres despejar aparece \textbf{al cuadrado} y además \textbf{linealmente} (sumando o restando), el despeje no termina en una fórmula sencilla: termina en una \textbf{ecuación de segundo grado}.

  \textbf{Forma general:}
  $$a\,x^2 + b\,x + c = 0, \qquad a \neq 0$$
  \textbf{Solución (fórmula general o resolvente):}
  $$x = \frac{-\,b \pm \sqrt{b^2 - 4ac}}{2a}$$
  Cuando los números son sencillos conviene \textbf{factorizar}: si $x^2 + b\,x + c = (x - r_1)(x - r_2)$, las soluciones son $x = r_1$ y $x = r_2$.
\end{cuadroteoria}

\begin{cuadropasos}
  \textbf{Cómo resolver un despeje que termina en una ecuación de segundo grado:}
  \begin{enumerate}[leftmargin=*]
    \item Despeja hasta dejar \emph{todo} de un mismo lado e igualado a cero.
    \item Ordena los términos: primero el cuadrado, luego el lineal, luego el número: $a\,x^2 + b\,x + c = 0$.
    \item Resuelve factorizando o aplicando la fórmula general.
    \item \textbf{Interpreta los resultados:} descarta las raíces sin sentido físico (un tiempo, una longitud, una masa o una rapidez negativos) y quédate con la(s) válida(s).
    \item Sustituye ese valor en la fórmula que pedía el ejercicio y responde con su unidad.
  \end{enumerate}
\end{cuadropasos}

\begin{cuadroadvertencia}
  \textbf{Errores clásicos con las ecuaciones de segundo grado:}
  \begin{itemize}[leftmargin=*]
    \item \textbf{Perder una raíz:} no dividas entre la variable antes de factorizar. De $t^2 - 20t = 0$ se obtiene $t\,(t - 20) = 0$, es decir $t = 0$ y $t = 20$; si divides entre $t$, pierdes una solución.
    \item \textbf{Olvidar el signo:} en la fórmula general, $b$ incluye su signo (por eso aparece $-\,b$).
    \item \textbf{Descartar una raíz sin analizarla:} a veces \emph{las dos} soluciones son válidas; por ejemplo, los dos instantes en que un objeto pasa por la misma altura.
    \item \textbf{Confundir el discriminante $b^2 - 4ac$:} si es negativo, no hay solución real (revisa el despeje); si es cero, la solución es única.
  \end{itemize}
\end{cuadroadvertencia}

\subsection{Ejemplos resueltos}

\textbf{Ejemplo R1 (fácil).} De $v = \dfrac{d}{t}$ despeja $t$ y calcula su valor si $v = 15$ m/s y $d = 300$ m.

\begin{cuadrosolucion}
  \textbf{Solución:} la $t$ está dividiendo; la pasamos multiplicando:
  $$v \cdot t = d \qquad \Longrightarrow \qquad t = \frac{d}{v}$$
  Evaluando: $t = \dfrac{300 \text{ m}}{15 \text{ m/s}} = 20 \text{ s}$.
\end{cuadrosolucion}

\textbf{Ejemplo R2 (fácil).} De $\rho = \dfrac{m}{V}$ despeja $m$ y calcula su valor si $\rho = 1{,}5$ g/cm$^3$ y $V = 20$ cm$^3$.

\begin{cuadrosolucion}
  \textbf{Solución:} la $V$ divide; pasa multiplicando:
  $$m = \rho \cdot V = 1{,}5 \frac{\text{g}}{\text{cm}^3} \cdot 20 \text{ cm}^3 = 30 \text{ g}.$$
\end{cuadrosolucion}

\textbf{Ejemplo R3 (medio).} De la conversión $C = \dfrac{5}{9}(F - 32)$ despeja $F$ y calcula $F$ si $C = 25$.

\begin{cuadrosolucion}
  \textbf{Solución:} de afuera hacia adentro.
  \begin{enumerate}[leftmargin=*, label=\arabic*.]
    \item El $\frac{5}{9}$ multiplica al paréntesis: pasa dividiendo. $\dfrac{9}{5}C = F - 32$.
    \item El $32$ resta: pasa sumando. $\;F = \dfrac{9}{5}C + 32$.
  \end{enumerate}
  Evaluando: $F = \dfrac{9}{5}(25) + 32 = 45 + 32 = 77^{\circ}$F.
\end{cuadrosolucion}

\textbf{Ejemplo R4 (medio).} De $v_f = v_0 + a\,t$ despeja $a$ y calcula su valor si $v_f = 20$ m/s, $v_0 = 8$ m/s y $t = 3$ s.

\begin{cuadrosolucion}
  \textbf{Solución:} la $v_0$ suma, pasa restando; luego la $t$ multiplica, pasa dividiendo:
  $$a = \frac{v_f - v_0}{t} = \frac{20 - 8}{3} = \frac{12}{3} = 4 \frac{\text{m}}{\text{s}^2}.$$
\end{cuadrosolucion}

\textbf{Ejemplo R5 (desafiante).} De $T = 2\pi\sqrt{\dfrac{\ell}{g}}$ despeja $\ell$ y calcula su valor si $g = 9{,}8$ m/s$^2$ y $T = 4$ s.

\begin{cuadrosolucion}
  \textbf{Solución:}
  \begin{enumerate}[leftmargin=*, label=\arabic*.]
    \item El $2\pi$ multiplica: pasa dividiendo. $\dfrac{T}{2\pi} = \sqrt{\dfrac{\ell}{g}}$.
    \item Hay una raíz: elevamos al cuadrado ambos lados. $\dfrac{T^2}{4\pi^2} = \dfrac{\ell}{g}$.
    \item La $g$ divide: pasa multiplicando. $\ell = \dfrac{g\,T^2}{4\pi^2}$.
  \end{enumerate}
  Evaluando: $\ell = \dfrac{9{,}8 \cdot 4^2}{4\pi^2} = \dfrac{156{,}8}{39{,}48} \approx 3{,}97 \approx 4{,}0$ m.
\end{cuadrosolucion}

\textbf{Ejemplo R6 (muy difícil).} De $v^2 = v_0^2 + 2a\,\Delta x$ despeja $v_0$ y calcula su valor si $v = 25$ m/s, $a = 3$ m/s$^2$ y $\Delta x = 80$ m.

\begin{cuadrosolucion}
  \textbf{Solución:}
  \begin{enumerate}[leftmargin=*, label=\arabic*.]
    \item El término $2a\,\Delta x$ suma: pasa restando. $v_0^2 = v^2 - 2a\,\Delta x$.
    \item El cuadrado se elimina con raíz cuadrada (tomamos la raíz positiva, porque $v_0$ es una rapidez):
          $$v_0 = \sqrt{v^2 - 2a\,\Delta x}.$$
  \end{enumerate}
  Evaluando: $v_0 = \sqrt{25^2 - 2(3)(80)} = \sqrt{625 - 480} = \sqrt{145} \approx 12{,}0$ m/s.

  \textbf{Verificación:} $v_0^2 + 2a\,\Delta x = 145 + 480 = 625 = 25^2$. \checkmark
\end{cuadrosolucion}

\textbf{Ejemplo R7 (muy difícil).} Un objeto se lanza hacia arriba con $v_0 = 25$ m/s. Su altura cumple $h = v_0\,t - \dfrac{1}{2}g\,t^2$ con $g = 10$ m/s$^2$. Sustituye los datos y determina en qué instantes el objeto está a $30$ m de altura.

\begin{cuadrosolucion}
  \textbf{Solución:} sustituimos $h = 30$ m y llevamos todo a un mismo lado:
  $$30 = 25t - 5t^2 \qquad \Longrightarrow \qquad 5t^2 - 25t + 30 = 0 \qquad \Longrightarrow \qquad t^2 - 5t + 6 = 0$$
  Factorizando: $(t - 2)(t - 3) = 0$, de donde $t = 2$ s y $t = 3$ s.

  \textbf{Interpretación:} a los $2$ s el objeto \emph{sube} pasando por los $30$ m y a los $3$ s \emph{baja} pasando otra vez por esa altura. Las \textbf{dos raíces son válidas} porque ambas son tiempos positivos. (Con la fórmula general: $t = \dfrac{5 \pm \sqrt{25 - 24}}{2} = \dfrac{5 \pm 1}{2}$.)
\end{cuadrosolucion}

\textbf{Ejemplo R8 (muy difícil).} De $|\vec{v}| = \sqrt{v_x^2 + v_y^2}$ con $|\vec{v}| = 5$ y $v_y = v_x + 1$, despeja $v_x$ y calcula las componentes.

\begin{cuadrosolucion}
  \textbf{Solución:} sustituimos la relación entre las componentes:
  $$5 = \sqrt{v_x^2 + (v_x + 1)^2} \qquad \Longrightarrow \qquad 25 = v_x^2 + v_x^2 + 2v_x + 1 \qquad \Longrightarrow \qquad 2v_x^2 + 2v_x - 24 = 0$$
  Dividiendo entre $2$: $v_x^2 + v_x - 12 = 0$, que factoriza como $(v_x + 4)(v_x - 3) = 0$. Así, $v_x = 3$ o $v_x = -4$.

  \textbf{Descartamos} $v_x = -4$ porque exigiría $v_y = -3$ (vector en el tercer cuadrante, contrario al sentido pedido); entonces $v_x = 3$ y $v_y = 4$. \textbf{Verificación:} $\sqrt{3^2 + 4^2} = \sqrt{25} = 5$. \checkmark
\end{cuadrosolucion}

\end{enumerate}
