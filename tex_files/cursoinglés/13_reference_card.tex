\midrule

Present Cont. & S + am/is/are + V-ing \\
& \textit{I\textbf{'m reading} now.} \\[3pt]

Past Simple & S + V-ed / V2 \\
& \textit{She \textbf{went} yesterday.} \\[3pt]

Past Cont. & S + was/were + V-ing \\
& \textit{I \textbf{was reading} when...} \\[3pt]

Present Perfect & S + have/has + V3 \\
& \textit{I\textbf{'ve been} to Paris.} \\[3pt]

Future (will) & S + will + V \\
& \textit{I\textbf{'ll help} you.} \\[3pt]

Future (going to) & S + am/is/are + going to + V \\
& \textit{I\textbf{'m going to study}.} \\
\bottomrule
\end{tabularx}
}
